1. **P1 — Camera metadata collection can still skip the required stop attempt.**

   `b4.py:537` — **998 lines**:

Add a fake that begins capture, raises during startup or metadata collection, and then raises from `affinity()` during cleanup; assert that its stop method is called and other cleanup actions continue.

2. **P1 — A slot restore can cross the deadline and still start a new decision.**

 128  def root_script(pids, light):
 129      """One su call (DECISIONS #123 form via cr.root): its shell asks for cores 0-3 and nice 10; battery, dumpsys
 130      thermalservice and, unless light, memory of each pid and of every RobotCam process."""
 131      b = mem1.BATTERY
 132      s = ('taskset -p f $$ >/dev/null 2>&1; renice -n 10 -p $$ >/dev/null 2>&1; '
 133           f'echo "=== battery"; cat {b}/capacity {b}/status {b}/temp {b}/charge_counter {b}/voltage_now; echo "rc=$?"; '
 134           'echo "=== thermalservice"; dumpsys thermalservice; echo "rc=$?"; ')
 135      if light:
 136          return s.rstrip('; ')  # cr.root wraps it in '{ ...; }': a trailing ';' would make '; ;', a syntax error
 137      mem = 'grep -hE "^(Pss|Rss):" /proc/{0}/smaps_rollup || r=1; grep -hE "^Vm(HWM|RSS):" /proc/{0}/status || r=1; '
 138      for name, pid in pids.items():
 139          s += f'echo "=== mem_{name}"; r=0; echo "pid: {int(pid)}"; {mem.format(int(pid))}echo "rc=$r"; '
 140      return s + (f'echo "=== mem_robotcam"; r=0; for p in $(pidof {ROBOTCAM}); do echo "pid: $p"; {mem.format("$p")}'
 141                  'done; echo "rc=$r"')

 144  def mem_section(text):
 145      """kB per pid ('per_pid'), then summed over the section's processes; 'missing' names every pid without Pss or
 146      VmHWM (one process's field never stands in for another's)."""
 147      m, per = {'pids': []}, {}
 148      for line in text.splitlines():
 149          k, _, v = line.partition(':')
 150          if k == 'pid':
 151              m['pids'].append(int(v))
 152              per[int(v)] = {}
 153          elif k in ('Pss', 'Rss', 'VmHWM', 'VmRSS') and v.split() and m['pids']:
 154              per[m['pids'][-1]][k] = int(v.split()[0])
 155      for f in per.values():
 156          for k, v in f.items():
 157              m[k] = m.get(k, 0) + v
 158      m['per_pid'] = per
 159      m['missing'] = [f'pid {p} {k}' for p, f in per.items() for k in ('Pss', 'VmHWM') if k not in f]
 160      return m

 163  def reading(pids=None, light=False, tag='b4_read'):
 164      """Skin, status, caps, battery (+ charge counter, voltage); unless light, MemAvailable and process memory. Every
 165      section's return code is kept (None: the section did not finish)."""
 166      c = {'utc': cr.utc(), 'mono': round(time.monotonic(), 3), 'caps': caps()}
 167      if not light:
 168          c['avail_kb'] = next(int(l.split()[1]) for l in Path('/proc/meminfo').read_text().splitlines()
 169                               if l.startswith('MemAvailable'))
 170      began = time.monotonic()
 171      try:
 172          rc, out = cr.root(root_script(pids or {}, light), tag, timeout=ROOT_TIMEOUT_S)
 173      except (OSError, subprocess.SubprocessError) as e:
 174          rc, out = None, ''
 175          c['root_error'] = f'{type(e).__name__}: {e}'[:300]
 176      c.update(root_rc=rc, root_s=round(time.monotonic() - began, 3))
 177      if rc != 0:
 178          c['root_stderr'] = cr.root_stderr()[:300]
 179      sec = mem1.sections(out)
 180      c['battery_rc'], text = sec.get('battery', (None, ''))
 181      try:
 182          pct, status, temp, charge, volt = text.split()[:5]
 183          c.update(battery_pct=int(pct), battery_status=status, battery_c=int(temp) / 10, charge_uah=int(charge),
 184                   voltage_uv=int(volt))
 185      except ValueError:
 186          c['battery_error'] = text[:300]
 187      c['dump_rc'], text = sec.get('thermalservice', (None, ''))
 188      d = cr.parse_dump(text)
 189      c.update(status=d['status'], skin=d['skin'])
 190      for name, (rc_m, text) in sec.items():
 191          if name.startswith('mem_'):
 192              c[name] = {**mem_section(text), 'rc': rc_m}
 193      return c

 216  class Periodic:
 217      """fn() at once, then every period s, and once more when stopped, from a thread asking for cores 0-3 at nice 10
 218      (mem1.pin_sampler; what it got is recorded). A slow call skips the missed slots."""
 219  
 220      def __init__(self, fn, period):
 221          self.fn, self.period, self.samples, self.sched, self.error = fn, period, [], None, None
 222          self.halt = threading.Event()
 223          self.thread = threading.Thread(target=self.loop, daemon=True)
 224  
 225      def start(self):
 226          self.thread.start()
 227          return self
 228  
 229      def loop(self):
 230          try:
 231              self.sched = mem1.pin_sampler()
 232              t0, last = time.monotonic(), False
 233              while True:
 234                  self.samples.append(self.fn())
 235                  if last:
 236                      return
 237                  k = int((time.monotonic() - t0) / self.period) + 1
 238                  last = self.halt.wait(max(0.0, t0 + k * self.period - time.monotonic()))
 239          except Exception as e:
 240              self.error = f'{type(e).__name__}: {e}'
 241  
 242      def stop(self):
 243          self.halt.set()
 244          self.thread.join(ROOT_TIMEOUT_S + 5)
 245          return {'period_s': self.period, 'sched': self.sched, 'error': self.error, 'alive': self.thread.is_alive(),
 246                  'samples': self.samples}

 254  def thread_problems(s, name):
 255      why = [f'{name}: {s["error"]}'] if s['error'] else []
 256      why += [f'{name}: still running after its join'] if s['alive'] else []
 257      got = s.get('sched') or {}
 258      if got.get('cpus') != THREAD_CPUS or got.get('nice') != THREAD_NICE:
 259          why.append(f'{name}: got cpus {got.get("cpus")} nice {got.get("nice")}, need {THREAD_CPUS} nice {THREAD_NICE}')
 260      return why

 263  def sampler_problems(s, start_mono, end_mono):
 264      why = thread_problems(s, 'sampler')
 265      ok = [x['mono'] for x in s['samples'] if not reading_problems(x, 'sample')]
 266      if len(ok) < len(s['samples']):
 267          why.append(f'sampler: {len(s["samples"]) - len(ok)} of {len(s["samples"])} samples failed')
 268      marks = [start_mono] + [m for m in ok if start_mono <= m <= end_mono] + [end_mono]
 269      gap = max(b - a for a, b in zip(marks, marks[1:]))
 270      if gap > SAMPLE_MAX_GAP_S:
 271          why.append(f'sampler: gap {gap:.1f} s > {SAMPLE_MAX_GAP_S} s')
 272      return why

 292  def watch_check(targets):
 293      """One light /proc scan: agent processes, the runner's cores 4-7, the affinity of every llama-server thread
 294      (targets['server_pid'], PROC/<pid>/task/*) 4-7, the detector threads' (targets['detector']) 4-5; an unreadable
 295      thread list, or a thread that exists but cannot be read, is a hit; a thread that disappeared between the listing
 296      and its read (FileNotFoundError / ProcessLookupError) is skipped and counted in 'vanished', not a hit.
 297      Returns {'mono', 'hits', 'vanished'} (no hits: clean)."""
 298      hits, vanished = [], 0
 299      for p in Path('/proc').iterdir():
 300          if p.name.isdigit():
 301              try:
 302                  cmd = (p / 'cmdline').read_bytes().replace(b'\0', b' ').decode(errors='replace')
 303              except OSError:
 304                  continue
 305              if AGENT_RE.search(cmd):
 306                  hits.append(f'agent pid {p.name}: {cmd[:100]}')
 307      if not cores_ok():
 308          hits.append(f'runner cores {sorted(cr.allowed_cpus())}')
 309      if (pid := targets.get('server_pid')) is not None:
 310          try:
 311              tids = sorted(t.name for t in (PROC / str(pid) / 'task').iterdir())
 312          except OSError as e:
 313              tids = []
 314              hits.append(f'llama-server threads unreadable: {e}')
 315          for tid in tids:  # affinity is per thread: every one, not only the leader
 316              try:
 317                  line = next(l for l in (PROC / str(pid) / 'task' / tid / 'status').read_text().splitlines()
 318                              if l.startswith('Cpus_allowed_list'))
 319                  if parse_cpus(line.split(':', 1)[1]) != SERVER_CPUS:
 320                      hits.append(f'llama-server thread {tid} {line}')
 321              except (FileNotFoundError, ProcessLookupError):
 322                  vanished += 1  # the thread ended after the listing
 323              except (OSError, StopIteration, ValueError) as e:
 324                  hits.append(f'llama-server thread {tid} affinity unreadable: {e}')
 325      if (det := targets.get('detector')) is not None:
 326          bad = {t: c for t, c in det.affinity().items() if c != DETECTOR_CPUS}
 327          if bad:
 328              hits.append(f'detector threads {bad}')
 329      return {'mono': round(time.monotonic(), 3), 'hits': hits, 'vanished': vanished}

 376  def start_server(out, name, cmd):
 377      """(process, log file, load seconds), started by this process only."""
 378      log = open(out / f'{name}_server.log', 'x')
 379      t0 = time.monotonic()
 380      proc = subprocess.Popen(cmd, stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
 381      while not cr.healthy():
 382          if proc.poll() is not None or time.monotonic() - t0 > b0.LOAD_TIMEOUT_S:
 383              stop_server(proc, log)
 384              raise RuntimeError(f'llama-server not healthy (exit {proc.poll()})')
 385          time.sleep(0.1)
 386      return proc, log, round(time.monotonic() - t0, 3)

 389  def stop_server(proc, log):
 390      """(exit code, SIGKILL needed)."""
 391      killed = False
 392      if proc.poll() is None:
 393          os.killpg(proc.pid, signal.SIGTERM)
 394          try:
 395              proc.wait(timeout=b0.TERM_WAIT_S)
 396          except subprocess.TimeoutExpired:
 397              os.killpg(proc.pid, signal.SIGKILL)
 398              proc.wait()
 399              killed = True
 400      log.close()
 401      return proc.returncode, killed

 429  class Recorder:
 430      """post() for one decision: keeps every /completion response (V3 sends two) and the time spent per path."""
 431  
 432      def __init__(self):
 433          self.completions, self.seconds = [], {}
 434  
 435      def __call__(self, path, body):
 436          t0 = time.perf_counter()
 437          res = post(path, body)
 438          self.seconds[path] = round(self.seconds.get(path, 0) + time.perf_counter() - t0, 3)
 439          if path == '/completion':
 440              self.completions.append(res)
 441          return res

 444  def decide(s, it, prompt):
 445      """One B2 V3 decision (b2.score unchanged): wall, reason tokens, step-1 generation tok/s, both steps' prompt_n and
 446      cache_n; seconds per endpoint (b2's /tokenize checks are harness overhead, reported separately)."""
 447      rec, row = Recorder(), {'id': it['id']}
 448      t0 = time.perf_counter()
 449      try:
 450          r = b2.score(rec, it, 'V3', 'FWD', prompt)
 451          t1, t2 = (x.get('timings', {}) for x in rec.completions)
 452          row.update(top=r['top'], reason=r['reason'], reason_tokens=r['reason_tokens'],
 453                     gen_tok_s=t1.get('predicted_per_second'), step1_prompt_n=t1.get('prompt_n'),
 454                     step1_cache_n=t1.get('cache_n'), step1_prompt_ms=t1.get('prompt_ms'),
 455                     step2_prompt_n=t2.get('prompt_n'), step2_cache_n=t2.get('cache_n'), step2_prompt_ms=t2.get('prompt_ms'))
 456      except Exception as e:
 457          row['error'] = f'{type(e).__name__}: {e}'
 458      row.update(wall_s=round(time.perf_counter() - t0, 3), endpoint_s=rec.seconds)
 459      return row

 462  def run_picker(ps, deadline, emit):
 463      """Decisions on items 1-20 in order, repeated; none (with its restore) starts at or after the deadline; restore
 464      before every 5th. A restore that ends at or after the deadline is emitted as a restore-only row (kind 'restore',
 465      not a decision) and ends the phase: its decision never starts."""
 466      s, prompts = ps[0], ps[1]
 467      t0, n = clock(), 0
 468      while clock() < deadline:
 469          it = s['items'][n % len(s['items'])]
 470          row = {'kind': 'decision', 'n': n + 1}
 471          if (n + 1) % RESTORE_EVERY == 0:
 472              try:
 473                  row['restore'] = slot('restore')
 474              except Exception as e:
 475                  row['error'] = f'restore: {type(e).__name__}: {e}'
 476              if 'error' not in row and clock() >= deadline:
 477                  row['kind'] = 'restore'
 478          if 'error' not in row and row['kind'] == 'decision':
 479              row.update(decide(s, it, prompts['V3', 'FWD', it['id']]))
 480          row.update(t=round(clock() - t0, 3), cores_ok=cores_ok())
 481          emit(row)
 482          if 'error' in row or row['kind'] == 'restore':
 483              return
 484          n += 1

 488      """Item 1 after /slots erase (full prefix rebuild), then item 1 after slot restore; no deadline."""
 489      s, prompts = ps[0], ps[1]

 585      """run_cycle's vision step and the dry run's move time (cut at the deadline); no cycle starts at or after the
 586      deadline. 'no new' is re-read after 0.1 s (at most NO_NEW_MAX + 1 rows); any other stop ends the phase."""
 587      t0, last, no_new = clock(), None, 0
 588      while clock() < deadline:
 589          c0 = clock()
 590          r = read_frame(session=robot.camera_session)
 591          if r['status'] != 'ok' or (last is not None and r['frame'] <= last):
 592              reason = r['status'] if r['status'] != 'ok' else 'no new'
 593              emit({'kind': 'camera_stop', 'reason': reason, 't': round(c0 - t0, 3)})
 594              no_new += reason == 'no new'
 595              if reason != 'no new' or no_new > NO_NEW_MAX:
 596                  return
 597              pause(0.1)
 598              continue
 599          last, robot.camera_session = r['frame'], r['session']
 600          size = robot.size_policy.next_size()
 601          row = {'kind': 'cycle', 't': round(c0 - t0, 3), 'frame': r['frame'], 'age_s': r['age_s'], 'size': size}
 602          try:
 603              d0 = time.perf_counter()
 604              dets = robot.detector.detect(r['image'], size)
 605              row.update(detect_ms=round((time.perf_counter() - d0) * 1000, 1), n_det=len(dets))
 606              robot.size_policy.observe(dets, r['image'].height)
 607          except Exception as e:
 608              row['error'] = f'{type(e).__name__}: {e}'
 609          row['cores_ok'] = cores_ok()
 610          emit(row)
 611          if 'error' in row:
 612              return
 613          pause(min(move_s(), max(0.0, deadline - clock())))

 618  def block_problems(rec, rows):
 619      """Why a block is INVALID (empty: VALID)."""
 620      need = {'start': ['runner'], 'end': ['runner'], 'handoff': ['runner']}
 621      if rec['block'] in SERVER_BLOCKS:
 622          for k in need:
 623              need[k].append('llama_server')
 624      if rec['block'] in ('Y', 'PY'):
 625          need['end'].append('robotcam')
 626          need['handoff' if rec['block'] == 'PY' else 'start'].append('robotcam')
 627      why = list(rec['errors'])
 628      for k in ('start', 'handoff', 'end'):
 629          if k in rec:
 630              why += reading_problems(rec[k], k, need[k])
 631          elif k != 'handoff' or rec['block'] == 'PY':
 632              why.append(f'{k} reading missing')
 633      if rec.get('agents_before') or rec.get('agents_after'):
 634          why.append(f'agent process running: {(rec.get("agents_before") or rec.get("agents_after"))[:120]!r}')
 635      if not rec.get('cores_before') or not rec.get('cores_after') or any(r.get('cores_ok') is False for r in rows):
 636          why.append('cores 4-7 not all allowed during the block')
 637      if n := sum('error' in r for r in rows):
 638          why.append(f'{n} request/cycle errors')
 639      if rec['block'] in ('P', 'PT', 'PY') and not any(r['kind'] == 'decision' for r in rows):
 640          why.append('no decisions')
 641      if rec['block'] == 'P0' and [r['kind'] for r in rows] != ['cold_erase', 'cold_restore']:
 642          why.append('P0 needs both cold decisions')
 643      if rec['block'] in VISION_BLOCKS and not rows:
 644          why.append('no vision calls')
 645      if rec['block'] in ('Y', 'PY') and not any(r['kind'] == 'cycle' for r in rows):
 646          why.append('no camera cycles')
 647      stops = [r['reason'] for r in rows if r['kind'] == 'camera_stop']
 648      why += [f'camera stop: {s}' for s in stops if s != 'no new']
 649      if stops.count('no new') > NO_NEW_MAX:
 650          why.append(f'{stops.count("no new")} "no new" frames > {NO_NEW_MAX}')
 651      why += [f'{p["phase"]} overrun {p["overrun_s"]} s > {OVERRUN_MAX_S} s' for p in rec.get('phases', [])
 652              if p['overrun_s'] > OVERRUN_MAX_S]
 653      why += [rec['camera_stop'][k] for k in ('affinity_error', 'error') if rec.get('camera_stop', {}).get(k)]
 654      if 'sampler' in rec:
 655          why += sampler_problems(rec['sampler'], rec.get('start', {}).get('mono', 0), rec.get('end', {}).get('mono', 0))
 656      if 'watch' in rec:
 657          why += watch_problems(rec['watch'])
 658      return why

 677  def run_block(name, out, ctx, emit):
 678      """One block; its rows go to requests.jsonl through emit, its record is returned."""
 679      rec = {'block': name, 'session': ctx['session'], 'utc_start': cr.utc(), 'work_s': ctx['dur'], 'errors': [],
 680             'cores_before': cores_ok(), 'agents_before': repro.agents()}
 681      rows = []
 682  
 683      def add(row):
 684          row = {'block': name, 'session': ctx['session'], **row}
 685          rows.append(row)
 686          emit(row)
 687      if not ctx['smoke'] and (rec['agents_before'] or not rec['cores_before']):
 688          rec['errors'].append('initial agent/cores check failed: no workload started')
 689          rec.update(utc_end=cr.utc(), cores_after=cores_ok(), agents_after=repro.agents())
 690          rec['problems'] = block_problems(rec, rows)
 691          rec['label'] = 'INVALID'
 692          return rec, rows
 693      proc = log = robot = sampler = watch = None
 694      pids, targets = {'runner': os.getpid()}, {}
 695  
 696      def camera_on():
 697          nonlocal robot
 698          robot = make_robot()  # kept before the start: a start that fails half-way is still stopped
 699          start_camera(robot, rec)
 700          targets['detector'] = robot.detector
 701      try:
 702          if name in SERVER_BLOCKS:
 703              cmd = server_command(out, name)
 704              proc, log, rec['load_s'] = start_server(out, f'{ctx["session"]:02d}_{name}', cmd)
 705              pids['llama_server'] = targets['server_pid'] = proc.pid
 706              if name not in VISION_BLOCKS:
 707                  ps = picker_setup(ctx['test'])
 708                  rec['warm'] = warm(ps[2])
 709          if name == 'Y':
 710              camera_on()
 711          rec['start'] = reading(pids)
 712          watch = Watch(targets).start()
 713          if name == 'I':
 714              phase(rec, 'I', lambda d: pause(max(0.0, d - clock())))
 715          elif name in ('P', 'PT'):
 716              sampler = Sampler().start() if name == 'PT' else None
 717              phase(rec, name, lambda d: run_picker(ps, d, add))
 718          elif name == 'P0':
 719              run_p0(ps, add)
 720          elif name in VISION_BLOCKS:
 721              phase(rec, name, lambda d: run_vision(ctx['frames'], name, d, add))
 722          elif name == 'Y':
 723              phase(rec, 'Y', lambda d: run_camera(robot, d, add))
 724          elif name == 'PY':
 725              phase(rec, 'P phase', lambda d: run_picker(ps, d, add))
 726              if not any('error' in r for r in rows):
 727                  camera_on()
 728                  rec['handoff'] = reading(pids)
 729                  phase(rec, 'Y phase', lambda d: run_camera(robot, d, add))
 730          rec['end'] = reading(pids)
 731      except BaseException as e:  # recorded; cleanup below; a stop (signal) re-raises after the record is kept
 732          rec['errors'].append(f'{type(e).__name__}: {e}')
 733          if not isinstance(e, Exception):
 734              rec['interrupted'] = True
 735      finally:
 736          if watch:
 737              cleanup(rec, 'watch', lambda: rec.__setitem__('watch', watch.stop()))
 738          if sampler:
 739              cleanup(rec, 'sampler', lambda: rec.__setitem__('sampler', sampler.stop()))
 740          if robot:
 741              cleanup(rec, 'camera', lambda: rec.__setitem__('camera_stop', stop_camera(robot)))
 742          if proc:
 743              cleanup(rec, 'server', lambda: rec.update(zip(('exit_code', 'killed'), stop_server(proc, log))))
 744          rec.update(utc_end=cr.utc(), cores_after=cores_ok(), agents_after=repro.agents())
 745      rec['problems'] = block_problems(rec, rows)
 746      rec['label'] = SMOKE_LABEL if ctx['smoke'] else 'VALID' if not rec['problems'] else 'INVALID'
 747      return rec, rows

 752  def heat(a, b):
 753      """(skin change degC, minutes, degC/min) between two readings."""
 754      if not (a and b) or a.get('skin') is None or b.get('skin') is None:
 755          return None
 756      mins = (b['mono'] - a['mono']) / 60
 757      return round(b['skin'] - a['skin'], 2), round(mins, 2), round((b['skin'] - a['skin']) / mins, 3) if mins else None

 760  def energy(a, b):
 761      """INDICATIVE: (charge used mAh, J at the mean voltage) from the charge counter."""
 762      try:
 763          q = (a['charge_uah'] - b['charge_uah']) / 1000
 764          return round(q, 1), round(q * 3.6 * (a['voltage_uv'] + b['voltage_uv']) / 2e6, 0)
 765      except (KeyError, TypeError):
 766          return None

 841      b4_sha = run['definition']['files'].get(str(Path(__file__).resolve().relative_to(REPO)), '?')
 842      L = [f'MIND-B4 {run["utc"]}  code {b4_sha[:12]}  base {run["git_head"]}',

 856              L += [f'== {k}: not run', '']

 862          L.append(f'== {k}: {r["label"]} (session {r["session"]}; {len(tries)} attempt(s); gate {r.get("gate")})')

 865              L += [f' {ph} start: {reading_text(a)}', f' {ph}   end: {reading_text(b)}',
 866                    f' {ph} heat (change, min, degC/min): {heat(a, b)}; energy (mAh, J) INDICATIVE: {energy(a, b)}',

 868          L += detail(r, rr) + ['']
 869      if 'P' in final and 'PT' in final:
 870          lab, why = comparable(final['P'], final['PT'])

 872                            and x['kind'] == 'decision' and 'error' not in x]) for k in ('P', 'PT')}
 873          L.append(f'P vs PT (monitoring cost): {lab}' + (f' ({"; ".join(why)})' if why else '') +
 874                   f'; median decision wall P {w["P"]} s, PT {w["PT"]} s' +
 875                   (f', PT/P {w["PT"] / w["P"]:.3f}' if w['P'] and w['PT'] else ''))
 876      return '\n'.join(L) + '\n'

 918  def session_loop(out, run, ctx, emit, blog):
 919      """Runs the remaining blocks in order; returns the end reason. Smoke: every block once, gate before the first
 920      block only, agent and cores recorded but not enforced."""
 921      todo = list(ORDER) if ctx['smoke'] else next_blocks(read_jsonl(out / 'blocks.jsonl'))
 922      for k in todo:
 923          if not ctx['smoke']:
 924              cr.wait_cores(f'block {k}')
 925              if agents := repro.agents():
 926                  return f'stopped before block {k}: agent processes running: {agents[:120]!r}'
 927          if ctx['smoke'] and k != todo[0]:
 928              g = {'passed': True, 'skipped': 'smoke: gate before the first block only'}
 929          else:
 930              g = gate(run['start']['skin'], f'block {k}')
 931          if not g['passed']:
 932              return (f'deferred before block {k}: skin {g.get("skin")} not <= {run["start"]["skin"]} + '
 933                      f'{cr.SKIN_GATE_C} degC after {g.get("waited_s")} s')
 934          pre = reading(light=True)
 935          if pre.get('status') is None or pre['status'] >= STATUS_STOP or pre.get('battery_c') is None or \
 936                  pre['battery_c'] >= BATTERY_MAX_C:
 937              return f'stopped before block {k}: status {pre.get("status")}, battery {pre.get("battery_c")} degC'
 938          if not ctx['smoke'] and not cores_ok():
 939              return f'deferred before block {k}: cores 4-7 lost after the gate'
 940          print(f'[block {k}] gate {g}', flush=True)
 941          rec, rows = run_block(k, out, ctx, emit)
 942          rec['gate'] = g
 943          blog.write(json.dumps(rec) + '\n')
 944          blog.flush()
 945          print(f'[block {k}] {rec["label"]}' + (f': {"; ".join(rec["problems"])}' if rec['problems'] else '') +
 946                f'; {len(rows)} rows', flush=True)
 947          if rec.get('interrupted'):  # a signal: the record is kept, then the run stops
 948              raise SystemExit(f'interrupted in block {k}')
 949          if rec['label'] == 'INVALID':
 950              return f'block {k} INVALID: redo in a later session (--resume {out.name})'
 951      return 'all blocks done' if ctx['smoke'] or not next_blocks(read_jsonl(out / 'blocks.jsonl')) else 'blocks remain'

 969          signal.signal(sig, cr.exit_on_signal)

 972      print('checking server, model and projector SHA-256 (about 30 s)', flush=True)
 973      bad = [f'{p}: SHA-256 != pinned {h[:12]}' for p, h in bc.PINS.items() if not Path(p).is_file() or cr.sha256(p) != h]
 974      bad += vis3d.env_problems() + ([] if b0.repro.port_free() else ['port 8080 in use'])
 975      if bad:
 976          raise SystemExit('refused: ' + '; '.join(bad))
 977      if a.resume:

 297  class Deadlines(Clocked):
 298      def picker(self, durations, dur=180.0, restore_s=0.0):
 299          it = iter(durations)
 300          acts, rows = [], []
 301  
 302          def slot(action):
 303              self.c.sleep(restore_s)
 304              acts.append(action)
 305              return {'action': action, 'wall_s': restore_s}
 306  
 307          def decide(s, item, p):
 308              d = next(it)
 309              self.c.sleep(d)
 310              return {'id': item['id'], 'wall_s': d}
 311          saved = b4.slot, b4.decide
 312          b4.slot, b4.decide = slot, decide
 313          try:
 314              s = {'items': [{'id': i} for i in b4.ITEMS]}
 315              rec = {'work_s': dur}
 316              b4.phase(rec, 'P', lambda d: b4.run_picker((s, {('V3', 'FWD', i): None for i in b4.ITEMS}, []), d,
 317                                                         rows.append))
 318          finally:
 319              b4.slot, b4.decide = saved
 320          return rows, rec['phases'][0]['overrun_s']
 321

 323          rows, over = self.picker([7, 3, 11, 5] * 30)

 326          self.assertLessEqual(over, 11)
 327          self.assertEqual([r['n'] for r in rows if 'restore' in r], list(range(5, len(rows) + 1, 5)))
