   255	    same = [b for b in blocks if b["block"] != "main" and b["threads"] == main["threads"]]
   256	    thread_choice_diffs, thread_prob_diff = diffs(other)
   257	    nondet_choice_diffs, nondet_prob_diff = diffs(same)
   258	    cached = next((b for b in blocks if b["block"] == f"t{main['threads']}"), None)
   259	    # Same decision, same thread count, two blocks: their ratio shows how disturbed the timings were.
   260	    repeat = [(r["case_id"], main_written[r["case_id"]]["total_ms"], r["total_ms"]) for r in (cached or {"rows": []})["rows"]]
   261	    ratio = lambda a, b: max(a, b) / max(1e-6, min(a, b))  # epsilon: a 0.0 ms reading must not divide by zero
   262	    worst = max(repeat, key=lambda x: ratio(x[1], x[2]), default=None)
   263	    load = lambda b: None if b is None else {**b["phases"], "total": sum(b["phases"].values()),
   264	                                              "spawn_to_ready": b["spawn_to_ready_ms"], "file_bytes": b["file_bytes"],
   265	                                              "state": b["load_state"]}
   266	    return {
   267	        "model": name, "threads_main": main["threads"], "cases": len(written),
   268	        "correct": rate(rows, "correct"), "acceptable": rate(rows, "acceptable"),
   269	        "by_level": by("level"), "by_phrasing": by("phrasing"),
   270	        "flips": sum(written[c] != reverse[c] for c in written), "flip_cases": [c for c in written if written[c] != reverse[c]],
   271	        "median_ms": median(ms), "p95_ms": p95(ms),
   272	        "tok_per_s": median([r["tokens"] / (max(1e-6, r["total_ms"]) / 1000) for r in rows]),
   273	        "split_ms": {k: median([r[k] for r in rows]) for k in ("tokenise_ms", "forward_ms", "post_ms")},
   274	        "split_share": {k: sum(r[k] for r in rows) / max(1e-6, sum(ms)) for k in ("tokenise_ms", "forward_ms", "post_ms")},
   275	        "fit": fit([r["tokens"] for r in rows], ms),
   276	        "tokens_by_level": {L: median([r["tokens"] for r in rows if r["level"] == L]) for L in levels},
   277	        "truncated": sum(r["longest"] >= r["limit"] for r in rows), "limit": rows[0]["limit"],
   278	        "scaling": scaling, "repeat_worst": worst and {"case_id": worst[0], "main_ms": worst[1], "repeat_ms": worst[2],
   279	                                                        "ratio": ratio(worst[1], worst[2])},
   280	        "thread_choice_diffs": thread_choice_diffs, "thread_max_prob_diff": thread_prob_diff,
   281	        "nondet_choice_diffs": nondet_choice_diffs, "nondet_max_prob_diff": nondet_prob_diff,
   282	        "nondet_n": sum(len(b["rows"]) for b in same), "thread_n": sum(len(b["rows"]) for b in other),
   283	        "cold": load(main), "cached": load(cached),
   284	        "peak_mib": max(b["peak_mib"] for b in blocks), "wait4_maxrss_mib": max(b["wait4_maxrss_mib"] or 0 for b in blocks),
   285	        "timing_note": main["timing_note"], "token_def": main["token_def"], "torch_threads": {b["block"]: b["torch_threads"] for b in blocks},
   286	    }

   291	def hints(m, mem_total_mib):
   292	    out = []
   293	    pct = lambda x: f"{x * 100:.0f}%"
   294	    f = m["fit"]
   295	    if f is None:
   296	        out.append(f"no latency-vs-length hint: all decisions had the same token count "
   297	                   f"({sorted(set(m['tokens_by_level'].values()))}), nothing to fit")
   298	    elif f["r2"] < 0.5:
   299	        out.append(f"latency is poorly explained by input length (R2 {f['r2']:.2f}, n {f['n']}, tokens "
   300	                   f"{f['tokens_min']}-{f['tokens_max']}) -> shorter input is not a reliable speed-up")
   301	    elif f["per_token_ms"] <= 0:
   302	        out.append(f"no measurable per-token cost (slope {f['per_token_ms']:.3f} ms/token, R2 {f['r2']:.2f}) "
   303	                   "-> shorter input will not help")
   304	    else:
   305	        top = max(m["tokens_by_level"])
   306	        t = m["tokens_by_level"][top]
   307	        # a negative fitted fixed term would push this past 100%; cap it, the evidence shows the term
   308	        share = min(1.0, f["per_token_ms"] * t / max(1e-6, f["fixed_ms"] + f["per_token_ms"] * t))
   309	        ev = (f"fit {f['fixed_ms']:.0f} ms + {f['per_token_ms']:.2f} ms/token x {t:.0f} tokens at level {top}, "
   310	              f"R2 {f['r2']:.2f}")
   311	        if share >= 0.5:
   312	            out.append(f"per-token cost is {pct(share)} of predicted latency at level {top} ({ev}) -> shorter input will help")
   313	        else:
   314	            out.append(f"fixed cost is {pct(1 - share)} of predicted latency at level {top} ({ev}) -> shortening input "
   315	                       "saves little; fewer model calls will help more")
   316	    s, ms = m["split_share"], m["split_ms"]
   317	    split_ev = (f"median tokenise {ms['tokenise_ms']:.1f} / forward {ms['forward_ms']:.1f} / post {ms['post_ms']:.1f} ms; "
   318	                f"shares {pct(s['tokenise_ms'])}/{pct(s['forward_ms'])}/{pct(s['post_ms'])}")
   319	    if s["tokenise_ms"] >= 0.2:
   320	        out.append(f"tokenisation is {pct(s['tokenise_ms'])} of decision time ({split_ev}) -> pre-tokenise constant "
   321	                   "parts or use a faster tokenizer")
   322	    if s["post_ms"] >= 0.2:
   323	        out.append(f"post-processing is {pct(s['post_ms'])} of decision time ({split_ev}) -> the glue code around the "
   324	                   "model is worth optimising")
   325	    if s["forward_ms"] >= 0.8:
   326	        out.append(f"model forward is {pct(s['forward_ms'])} of decision time ({split_ev}) -> only fewer tokens or a "
   327	                   "faster model will help; glue code is not the bottleneck")
   328	    sc = {t: v["median_ms"] for t, v in m["scaling"].items() if v["median_ms"] is not None}
   329	    rw = m["repeat_worst"]
   330	    if len(sc) >= 2 and rw and rw["ratio"] > 1.3:
   331	        out.append(f"no thread-count hint: the same {m['threads_main']}-thread decision took {rw['main_ms']:.0f} ms in the "
   332	                   f"main block and {rw['repeat_ms']:.0f} ms in the t{m['threads_main']} block ({rw['case_id']}, "
   333	                   f"{rw['ratio']:.1f}x; limit 1.3x) -> timings were disturbed (thermal or scheduler); rerun the scaling blocks")
   334	    elif len(sc) >= 2:
   335	        lo = min(sc)
   336	        best = min(sc, key=lambda t: (sc[t], t))  # fastest; a tie goes to fewer threads
   337	        ranked = " < ".join(f"{t} thr {sc[t]:.0f} ms" for t in sorted(sc, key=lambda t: (sc[t], t)))
   338	        ev = f"ranked {ranked} (median of {m['scaling'][lo]['n']} each)"
   339	        noise = max(0.15, rw["ratio"] - 1 if rw else 0)  # a gap below the repeat noise is not a finding
   340	        gain = (sc[lo] - sc[best]) / sc[lo]
   341	        if best == lo:
   342	            out.append(f"no thread count beats {lo} threads ({ev}) -> run at {lo} threads")
   343	        elif gain <= noise:
   344	            out.append(f"fastest ({best} threads) is only {pct(gain)} faster than {lo}, within the {pct(noise)} noise floor "
   345	                       f"({ev}) -> {lo} threads frees cores at no measurable cost")
   346	        else:
   347	            out.append(f"{best} threads is {pct(gain)} faster than {lo}, above the {pct(noise)} noise floor ({ev}) "
   348	                       f"-> run at {best} threads")
   349	    if m["nondet_choice_diffs"]:
   350	        out.append(f"run-to-run non-determinism: the same decision at the same {m['threads_main']} threads changed choice "
   351	                   f"in {m['nondet_choice_diffs']}/{m['nondet_n']} repeats (max probability difference "
   352	                   f"{m['nondet_max_prob_diff']:.4f}) -> results are not reproducible run to run")
   353	    if m["thread_choice_diffs"]:
   354	        if m["nondet_choice_diffs"]:
   355	            out.append(f"choice changed at other thread counts in {m['thread_choice_diffs']}/{m['thread_n']} decisions, "
   356	                       "but same-thread repeats also changed -> cannot separate a thread effect from run-to-run "
   357	                       "non-determinism")
   358	        else:
   359	            out.append(f"choice changed with thread count in {m['thread_choice_diffs']}/{m['thread_n']} decisions (max "
   360	                       f"probability difference {m['thread_max_prob_diff']:.4f}) while same-thread repeats matched "
   361	                       f"({m['nondet_n']}) -> results depend on thread count")
   362	    cold, cached = m["cold"], m["cached"]
   363	    if cold and cached:
   364	        d = cold["total"] - cached["total"]
   365	        if d >= 1000 and d >= 0.25 * cached["total"]:
   366	            out.append(f"{cold['state']['mode'].split(' ')[0]} load is {d / 1000:.1f} s slower than cached "
   367	                       f"({cold['total'] / 1000:.1f} vs {cached['total'] / 1000:.1f} s; file read "
   368	                       f"{cold['file_read'] / 1000:.1f} vs {cached['file_read'] / 1000:.1f} s) -> keep the worker resident "
   369	                       "or the weights in page cache")
   370	        for phase, advice in (("init", "a faster-loading format or a resident worker will help"),
   371	                              ("import", "library import dominates; a resident worker avoids it")):
   372	            share = cached[phase] / cached["total"]
   373	            if share >= 0.5:
   374	                out.append(f"{phase} is {pct(share)} of cached load ({cached[phase] / 1000:.1f} of "
   375	                           f"{cached['total'] / 1000:.1f} s) -> {advice}")
   376	        if m["median_ms"] and cached["warmup"] >= 2 * m["median_ms"] and cached["warmup"] - m["median_ms"] >= 200:
   377	            out.append(f"first call takes {cached['warmup']:.0f} ms vs median {m['median_ms']:.0f} ms, although the warm-up "
   378	                       "input is a short 2-option decision -> make a warm-up call at startup")
   379	    n = m["cases"]
   380	    if n < MIN_N:
   381	        out.append(f"order flips {m['flips']}/{n} cases: too few to judge (need >= {MIN_N} cases)")
   382	    elif m["flips"] / n >= 0.1:
   383	        out.append(f"choice changed under reversed order in {m['flips']}/{n} cases ({pct(m['flips'] / n)}) -> order bias; "
   384	                   "average over orders or fix the order")
   385	    lv = {L: v["correct"] for L, v in m["by_level"].items()}
   386	    few = {L: c[1] for L, c in lv.items() if c[1] < MIN_N}
   387	    if len(lv) >= 2 and (min(lv) in few or max(lv) in few):
   388	        out.append("level: " + ", ".join(f"L{L} {c[1]} decisions" for L, c in lv.items()) +
   389	                   f" -> too few to judge (need >= {MIN_N} decisions at the lowest and highest level)")
   390	    elif len(lv) >= 2:
   391	        a, b = min(lv), max(lv)
   392	        ra, rb = lv[a][0] / lv[a][1], lv[b][0] / lv[b][1]
   393	        if ra - rb >= 0.3:
   394	            out.append(f"accuracy falls from {pct(ra)} at level {a} ({lv[a][0]}/{lv[a][1]}) to {pct(rb)} at level {b} "
   395	                       f"({lv[b][0]}/{lv[b][1]}) -> the model breaks down at the harder levels")
   396	    ph = {p: v["correct"] for p, v in m["by_phrasing"].items() if v["correct"][1] >= MIN_N}
   397	    if len(m["by_phrasing"]) >= 2 and len(ph) < 2:
   398	        out.append("phrasing: " + ", ".join(f"{p} {v['correct'][1]} decisions" for p, v in m["by_phrasing"].items()) +
   399	                   f" -> too few to judge (need >= {MIN_N} decisions in at least two phrasings)")
   400	    elif len(ph) >= 2:
   401	        best, worst = max(ph, key=lambda p: ph[p][0] / ph[p][1]), min(ph, key=lambda p: ph[p][0] / ph[p][1])
   402	        gap = ph[best][0] / ph[best][1] - ph[worst][0] / ph[worst][1]
   403	        if gap >= 0.25:
   404	            out.append(f"sensitive to phrasing: '{best}' {ph[best][0]}/{ph[best][1]} vs '{worst}' {ph[worst][0]}/"
   405	                       f"{ph[worst][1]} correct ({pct(gap)} gap) -> normalise phrasing before the model")
   406	    if m["truncated"]:
   407	        out.append(f"{m['truncated']} decisions hit the {m['limit']}-token limit -> input was cut; shorten it")
   408	    if m["peak_mib"] >= 0.4 * mem_total_mib:
   409	        out.append(f"peak RAM {m['peak_mib']} MiB is {pct(m['peak_mib'] / mem_total_mib)} of device RAM "
   410	                   f"({mem_total_mib} MiB) -> only one such model can stay resident")
   411	    return out

   422	    p = L.append

   432	        p("  by phrasing   " + "  ".join(f"{k} {frac(v['correct'])}" for k, v in m["by_phrasing"].items()))
   433	        p(f"order flips     {m['flips']}/{m['cases']}" + (f"  {m['flip_cases']}" if m["flips"] else ""))
   434	        p(f"decision ms     median {m['median_ms']:.0f}  P95 {m['p95_ms']:.0f}  (main block, {m['threads_main']} threads, "
   435	          f"n={m['correct'][1]})")
   436	        sm = m["split_ms"]
   437	        p(f"  split median  tokenise {sm['tokenise_ms']:.1f}  forward {sm['forward_ms']:.1f}  post {sm['post_ms']:.1f}   "

   439	        p(f"tokens/s        {m['tok_per_s']:.0f}   tokens by level " +
   440	          "  ".join(f"L{k} {v:.0f}" for k, v in m["tokens_by_level"].items()) +
   441	          f"   limit {m['limit']}, hit {m['truncated']}x")
   442	        p(f"token count     {m['token_def']}")
   443	        f = m["fit"]
   444	        p("latency fit     " + (f"ms = {f['fixed_ms']:.1f} + {f['per_token_ms']:.3f} x tokens   R2 {f['r2']:.2f}  "
   445	                                f"(n={f['n']}, tokens {f['tokens_min']}-{f['tokens_max']})" if f else "not possible (no token spread)"))
   446	        p("threads         " + "  ".join(f"{t}: {v['median_ms']:.0f} ms" for t, v in sorted(m["scaling"].items())) +
   447	          f"  (written order, fresh cached worker, n={next(iter(m['scaling'].values()))['n']} each)"
   448	          f"   torch threads seen {m['torch_threads']}")
   449	        rw = m["repeat_worst"]
   450	        if rw:
   451	            p(f"                repeat check: worst {rw['case_id']} {rw['main_ms']:.0f} ms (main) vs {rw['repeat_ms']:.0f} ms "
   452	              f"(t{m['threads_main']}) = {rw['ratio']:.2f}x" + ("  DISTURBED (>1.3x)" if rw["ratio"] > 1.3 else ""))
   453	        p(f"                run-to-run non-determinism (main vs t{m['threads_main']}, same threads): choice differs in "
   454	          f"{m['nondet_choice_diffs']}/{m['nondet_n']}, max prob diff {m['nondet_max_prob_diff']:.4f}")
   455	        p(f"                other thread counts vs main: choice differs in {m['thread_choice_diffs']}/{m['thread_n']}, "
   456	          f"max prob diff {m['thread_max_prob_diff']:.4f}")
   457	        p(f"load (s)        {'phase':<10}{'cold':>8}{'cached':>8}")

   466	            p(f"                page cache {st['cached_mib_before']} -> {st['cached_mib_after']} MiB after the human drop")
   467	        p(f"RAM             peak {m['peak_mib']} MiB (VmHWM, worker{' + llama-server' if m['model'] == 's1o' else ''})   "

   469	        p("HINTS")
   470	        for h in m["hints"]:
   471	            p(f"  - {h}")
   472	    p("")
   473	    p("== CATEGORY WINNERS ==")
   474	    ms = res["per_model"]
   475	    def rank(label, key, fmt, reverse=False):
   476	        vals = [(key(m), m["model"]) for m in ms if key(m) is not None]
   477	        vals.sort(key=lambda v: v[0], reverse=reverse)
   478	        p(f"{label:<17}{vals[0][1]:<12} " + "  ".join(f"{n} {fmt(v)}" for v, n in vals))
   479	    rank("accuracy", lambda m: (m["correct"][0] / m["correct"][1], m["acceptable"][0] / m["acceptable"][1]),
   480	         lambda v: f"{v[0] * 100:.0f}%/{v[1] * 100:.0f}%acc", reverse=True)
   481	    rank("order stability", lambda m: m["flips"] / m["cases"], lambda v: f"{v * 100:.0f}% flips")
   482	    rank("decision speed", lambda m: m["median_ms"], lambda v: f"{v:.0f} ms")
   483	    rank("cached load", lambda m: m["cached"]["total"] if m["cached"] else None, lambda v: f"{v / 1000:.1f} s")
   484	    rank("cold load", lambda m: m["cold"]["total"], lambda v: f"{v / 1000:.1f} s")
   485	    rank("RAM", lambda m: m["peak_mib"], lambda v: f"{v} MiB")
   486	    p("(ties keep the listed model order; accuracy ties break on acceptable)")
   487	    return "\n".join(L)

   493	    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)

   514	           "levels": sorted({c["level"] for c in cases}), "models": models, "threads": threads,
   515	           "mem_total_mib": mem_total, "per_model": []}
   516	    with open(a.out / "decisions.jsonl", "x") as log:
   517	        for name in models:
   518	            files = files_of(WEIGHTS[name])
   519	            print(f"[{name}] main block: cold load, {len(cases)} cases x 2 orders at {threads[-1]} threads", flush=True)
   520	            blocks = [run_block(name, "main", threads[-1], cases, ("written", "reversed"), files, a.out, log, cold=True)]
   521	            for t in threads:
   522	                print(f"[{name}] t{t}: cached load, {len(cases)} cases written order", flush=True)
   523	                blocks.append(run_block(name, f"t{t}", t, cases, ("written",), files, a.out, log, cold=False))
   524	            m = analyse(name, blocks, cases)
   525	            m["hints"] = hints(m, mem_total)
   526	            m["blocks"] = [{k: v for k, v in b.items() if k != "rows"} for b in blocks]
   527	            res["per_model"].append(m)
   528	    order = {"CANDIDATE": 0, "OUT (speed)": 1, "DROPPED": 2}
   529	    res["per_model"].sort(key=lambda m: order[MODELS[m["model"]][3]])
   530	    (a.out / "results.json").write_text(json.dumps(res, indent=1, default=str) + "\n")
   531	    text = report(res)
   532	    (a.out / "report.txt").write_text(text + "\n")
   533	    print(text)
