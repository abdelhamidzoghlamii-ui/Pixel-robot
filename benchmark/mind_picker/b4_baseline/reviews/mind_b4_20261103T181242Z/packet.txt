 115  def mem_section(text):
 116      """kB summed over the section's processes, and their pids."""
 117      m = {'pids': []}
 118      for line in text.splitlines():
 119          k, _, v = line.partition(':')
 120          if k == 'pid':
 121              m['pids'].append(int(v))
 122          elif k in ('Pss', 'Rss', 'VmHWM', 'VmRSS') and v.split():
 123              m[k] = m.get(k, 0) + int(v.split()[0])
 124      return m

 166  class Sampler:
 167      """PT's root sampler: a reading(light) every period s from a thread on cores 0-3 at nice 10 (mem1.pin_sampler;
 168      what it got is recorded). A slow sample skips the missed slots."""
 169  
 170      def __init__(self, read=None, period=SAMPLE_S):
 171          self.read = read or (lambda: reading(light=True, tag='b4_sample'))
 172          self.period, self.samples, self.sched, self.error = period, [], None, None
 173          self.halt = threading.Event()
 174          self.thread = threading.Thread(target=self.loop, daemon=True)
 175  
 176      def start(self):
 177          self.thread.start()
 178          return self
 179  
 180      def loop(self):
 181          try:
 182              self.sched = mem1.pin_sampler()
 183              t0 = time.monotonic()
 184              while not self.halt.is_set():
 185                  self.samples.append(self.read())
 186                  k = int((time.monotonic() - t0) / self.period) + 1
 187                  self.halt.wait(max(0.0, t0 + k * self.period - time.monotonic()))
 188          except Exception as e:
 189              self.error = f'{type(e).__name__}: {e}'
 190  
 191      def stop(self):
 192          self.halt.set()
 193          self.thread.join(ROOT_TIMEOUT_S + 5)
 194          return {'period_s': self.period, 'sched': self.sched, 'error': self.error, 'alive': self.thread.is_alive(),
 195                  'samples': self.samples}

 198  def sampler_problems(s, start_mono, end_mono):
 199      why = [f'sampler: {s["error"]}'] if s['error'] else []
 200      why += ['sampler: still running after its join'] if s['alive'] else []
 201      ok = [x['mono'] for x in s['samples'] if not reading_problems(x, 'sample')]
 202      if len(ok) < len(s['samples']):
 203          why.append(f'sampler: {len(s["samples"]) - len(ok)} of {len(s["samples"])} samples failed')
 204      marks = [start_mono] + [m for m in ok if start_mono <= m <= end_mono] + [end_mono]
 205      gap = max(b - a for a, b in zip(marks, marks[1:]))
 206      if gap > SAMPLE_MAX_GAP_S:
 207          why.append(f'sampler: gap {gap:.1f} s > {SAMPLE_MAX_GAP_S} s')
 208      return why

 243  def start_server(out, name):
 244      """(process, log file, load seconds): bc.command, started by this process only."""
 245      log = open(out / f'{name}_server.log', 'x')
 246      t0 = time.monotonic()
 247      proc = subprocess.Popen(bc.command(out / 'slots'), stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT,
 248                              start_new_session=True)
 249      while not cr.healthy():
 250          if proc.poll() is not None or time.monotonic() - t0 > b0.LOAD_TIMEOUT_S:
 251              stop_server(proc, log)
 252              raise RuntimeError(f'llama-server not healthy (exit {proc.poll()})')
 253          time.sleep(0.1)
 254      return proc, log, round(time.monotonic() - t0, 3)

 257  def stop_server(proc, log):
 258      """(exit code, SIGKILL needed)."""
 259      killed = False
 260      if proc.poll() is None:
 261          os.killpg(proc.pid, signal.SIGTERM)
 262          try:
 263              proc.wait(timeout=b0.TERM_WAIT_S)
 264          except subprocess.TimeoutExpired:
 265              os.killpg(proc.pid, signal.SIGKILL)
 266              proc.wait()
 267              killed = True
 268      log.close()
 269      return proc.returncode, killed

 297  class Recorder:
 298      """post() for one decision: keeps every /completion response (V3 sends two) and the time spent per path."""
 299  
 300      def __init__(self):
 301          self.completions, self.seconds = [], {}
 302  
 303      def __call__(self, path, body):
 304          t0 = time.perf_counter()
 305          res = post(path, body)
 306          self.seconds[path] = round(self.seconds.get(path, 0) + time.perf_counter() - t0, 3)
 307          if path == '/completion':
 308              self.completions.append(res)
 309          return res

 312  def decide(s, it, prompt):
 313      """One B2 V3 decision (b2.score unchanged): wall, reason tokens, step-1 generation tok/s, both steps' prompt_n and
 314      cache_n; seconds per endpoint (b2's /tokenize checks are harness overhead, reported separately)."""
 315      rec, row = Recorder(), {'id': it['id']}
 316      t0 = time.perf_counter()
 317      try:
 318          r = b2.score(rec, it, 'V3', 'FWD', prompt)
 319          t1, t2 = (x.get('timings', {}) for x in rec.completions)
 320          row.update(top=r['top'], reason=r['reason'], reason_tokens=r['reason_tokens'],
 321                     gen_tok_s=t1.get('predicted_per_second'), step1_prompt_n=t1.get('prompt_n'),
 322                     step1_cache_n=t1.get('cache_n'), step1_prompt_ms=t1.get('prompt_ms'),
 323                     step2_prompt_n=t2.get('prompt_n'), step2_cache_n=t2.get('cache_n'), step2_prompt_ms=t2.get('prompt_ms'))
 324      except Exception as e:
 325          row['error'] = f'{type(e).__name__}: {e}'
 326      row.update(wall_s=round(time.perf_counter() - t0, 3), endpoint_s=rec.seconds)
 327      return row

 330  def run_picker(ps, dur, emit):
 331      """Decisions on items 1-20 in order, repeated, until the next would not fit in dur; restore before every 5th."""
 332      s, prompts = ps[0], ps[1]
 333      t0, n, last = time.monotonic(), 0, 0.0
 334      while n == 0 or time.monotonic() - t0 + last <= dur:
 335          it = s['items'][n % len(s['items'])]
 336          row = {'kind': 'decision', 'n': n + 1}
 337          if (n + 1) % RESTORE_EVERY == 0:
 338              try:
 339                  row['restore'] = slot('restore')
 340              except Exception as e:
 341                  row['error'] = f'restore: {type(e).__name__}: {e}'
 342          if 'error' not in row:
 343              row.update(decide(s, it, prompts['V3', 'FWD', it['id']]))
 344          row.update(t=round(time.monotonic() - t0, 3), cores_ok=cores_ok())
 345          emit(row)
 346          if 'error' in row:
 347              return
 348          last = row['wall_s'] + row.get('restore', {}).get('wall_s', 0)
 349          n += 1

 434  def run_camera(robot, dur, emit):
 435      """run_cycle's vision step and the dry run's move time; a frame that is not new is a camera stop."""
 436      import main
 437      t0, last = time.monotonic(), None

 454          except Exception as e:

 457          emit(row)
 458          if 'error' in row:

 465  def block_problems(rec, rows):
 466      """Why a block is INVALID (empty: VALID)."""
 467      need = {'start': ['runner'], 'end': ['runner'], 'handoff': ['runner']}
 468      if rec['block'] in SERVER_BLOCKS:
 469          for k in need:
 470              need[k].append('llama_server')
 471      if rec['block'] in ('Y', 'PY'):
 472          need['end'].append('robotcam')
 473          need['handoff' if rec['block'] == 'PY' else 'start'].append('robotcam')
 474      why = list(rec['errors'])
 475      for k in ('start', 'handoff', 'end'):
 476          if k in rec:
 477              why += reading_problems(rec[k], k, need[k])
 478          elif k != 'handoff' or rec['block'] == 'PY':
 479              why.append(f'{k} reading missing')
 480      if rec.get('agents_before') or rec.get('agents_after'):
 481          why.append(f'agent process running: {(rec.get("agents_before") or rec.get("agents_after"))[:120]!r}')
 482      if not rec.get('cores_before') or not rec.get('cores_after') or any(r.get('cores_ok') is False for r in rows):
 483          why.append('cores 4-7 not all allowed during the block')
 484      if n := sum('error' in r for r in rows):
 485          why.append(f'{n} request/cycle errors')
 486      if rec['block'] in ('P', 'PT', 'PY') and not any(r['kind'] == 'decision' for r in rows):
 487          why.append('no decisions')
 488      if rec['block'] == 'P0' and [r['kind'] for r in rows] != ['cold_erase', 'cold_restore']:
 489          why.append('P0 needs both cold decisions')
 490      if rec['block'] == 'V' and not rows:
 491          why.append('no vision calls')
 492      if rec['block'] in ('Y', 'PY') and not any(r['kind'] == 'cycle' for r in rows):
 493          why.append('no camera cycles')
 494      if rec.get('camera_stop', {}).get('error'):
 495          why.append(rec['camera_stop']['error'])
 496      if 'sampler' in rec:
 497          why += sampler_problems(rec['sampler'], rec.get('start', {}).get('mono', 0), rec.get('end', {}).get('mono', 0))
 498      return why

 501  def run_block(name, out, ctx, emit):
 502      """One block; its rows go to requests.jsonl through emit, its record is returned."""
 503      rec = {'block': name, 'session': ctx['session'], 'utc_start': cr.utc(), 'work_s': ctx['dur'], 'errors': [],
 504             'cores_before': cores_ok(), 'agents_before': repro.agents()}
 505      rows = []
 506  
 507      def add(row):
 508          row = {'block': name, 'session': ctx['session'], **row}
 509          rows.append(row)
 510          emit(row)
 511      proc = log = robot = sampler = None
 512      pids = {'runner': os.getpid()}
 513      try:
 514          if name in SERVER_BLOCKS:
 515              proc, log, rec['load_s'] = start_server(out, f'{ctx["session"]:02d}_{name}')
 516              pids['llama_server'] = proc.pid
 517              if name != 'V':
 518                  ps = picker_setup(ctx['test'])
 519                  rec['warm'] = warm(ps[2])
 520          if name == 'Y':
 521              robot = start_camera(rec)
 522          rec['start'] = reading(pids)
 523          if name == 'I':
 524              time.sleep(ctx['dur'])
 525          elif name in ('P', 'PT'):
 526              sampler = Sampler().start() if name == 'PT' else None
 527              run_picker(ps, ctx['dur'], add)
 528          elif name == 'P0':
 529              run_p0(ps, add)
 530          elif name == 'V':
 531              run_vision(ctx['frames'], ctx['dur'], add)
 532          elif name == 'Y':
 533              run_camera(robot, ctx['dur'], add)
 534          elif name == 'PY':
 535              run_picker(ps, ctx['dur'], add)
 536              if not any('error' in r for r in rows):
 537                  robot = start_camera(rec)
 538                  rec['handoff'] = reading(pids)
 539                  run_camera(robot, ctx['dur'], add)
 540          rec['end'] = reading(pids)
 541      except BaseException as e:  # recorded; cleanup below; a stop (signal) re-raises after the record is kept
 542          rec['errors'].append(f'{type(e).__name__}: {e}')
 543          if not isinstance(e, Exception):
 544              rec['interrupted'] = True
 545      finally:
 546          if sampler:
 547              rec['sampler'] = sampler.stop()
 548          if robot:
 549              rec['camera_stop'] = stop_camera(robot)
 550          if proc:
 551              rec['exit_code'], rec['killed'] = stop_server(proc, log)
 552          rec.update(utc_end=cr.utc(), cores_after=cores_ok(), agents_after=repro.agents())
 553      rec['problems'] = block_problems(rec, rows)
 554      rec['label'] = SMOKE_LABEL if ctx['smoke'] else 'VALID' if not rec['problems'] else 'INVALID'
 555      return rec, rows

 640  def report(run, recs, rows):
 641      L = [f'MIND-B4 {run["utc"]}  code {run["code_sha256"][Path(__file__).name][:12]}  base {run["git_head"]}',
 642           f'server: {" ".join(run["command"])}', f'launch-start skin {run["start"].get("skin")} degC (gate reference)']
 643      if run['smoke']:
 644          L.append(f'*** {SMOKE_LABEL}: {run["work_s"]} s blocks, no gate, checks not enforced ***')
 645      L += ['timings: VALID blocks only, from the owner run with no agent resident; energy INDICATIVE', '']
 646      final = {}
 647      for r in recs:
 648          if r['label'] in ('VALID', SMOKE_LABEL) or r['block'] not in final:
 649              final[r['block']] = r
 650      for k in ORDER:
 651          tries = [r for r in recs if r['block'] == k]
 652          if k not in final:
 653              L += [f'== {k}: not run', '']
 654              continue
 655          r = final[k]
 656          rr = [x for x in rows if x['block'] == k and x['session'] == r['session']]
 657          phases = [('', r.get('start'), r.get('end'))] if k != 'PY' else \
 658              [(' P phase', r.get('start'), r.get('handoff')), (' Y phase', r.get('handoff'), r.get('end'))]
 659          L.append(f'== {k}: {r["label"]} (session {r["session"]}; {len(tries)} attempt(s); gate {r.get("gate")})')
 660          L += [f'  problems: {"; ".join(r["problems"])}'] if r['problems'] else []
 661          for ph, a, b in phases:
 662              L += [f' {ph} start: {reading_text(a)}', f' {ph}   end: {reading_text(b)}',
 663                    f' {ph} heat (change, min, degC/min): {heat(a, b)}; energy (mAh, J) INDICATIVE: {energy(a, b)}',
 664                    f' {ph} RAM start: {mem_text(a)}', f' {ph} RAM   end: {mem_text(b)}']
 665          L += detail(r, rr) + ['']
 666      if 'P' in final and 'PT' in final:
 667          lab, why = comparable(final['P'], final['PT'])
 668          w = {k: mem1.med([x['wall_s'] for x in rows if x['block'] == k and x['session'] == final[k]['session']
 669                            and x['kind'] == 'decision' and 'error' not in x]) for k in ('P', 'PT')}
 670          L.append(f'P vs PT (monitoring cost): {lab}' + (f' ({"; ".join(why)})' if why else '') +
 671                   f'; median decision wall P {w["P"]} s, PT {w["PT"]} s' +
 672                   (f', PT/P {w["PT"] / w["P"]:.3f}' if w['P'] and w['PT'] else ''))
 673      return '\n'.join(L) + '\n'

 695  def session_loop(out, run, ctx, emit, blog):
 696      """Runs the remaining blocks in order; returns the end reason. Smoke: every block once, nothing enforced."""
 697      todo = list(ORDER) if ctx['smoke'] else next_blocks(read_jsonl(out / 'blocks.jsonl'))
 698      for k in todo:
 699          if not ctx['smoke']:
 700              cr.wait_cores(f'block {k}')
 701              if agents := repro.agents():
 702                  return f'stopped before block {k}: agent processes running: {agents[:120]!r}'
 703          g = gate(run['start']['skin'], f'block {k}', ctx['smoke'])
 704          if not g['passed']:
 705              return (f'deferred before block {k}: skin {g.get("skin")} not <= {run["start"]["skin"]} + '
 706                      f'{cr.SKIN_GATE_C} degC after {g.get("waited_s")} s')
 707          if not ctx['smoke']:
 708              pre = reading(light=True)
 709              if pre.get('status') is None or pre['status'] >= STATUS_STOP or pre.get('battery_c') is None or \
 710                      pre['battery_c'] >= BATTERY_MAX_C:
 711                  return f'stopped before block {k}: status {pre.get("status")}, battery {pre.get("battery_c")} degC'
 712              if not cores_ok():
 713                  return f'deferred before block {k}: cores 4-7 lost after the gate'
 714          print(f'[block {k}] gate {g}', flush=True)
 715          rec, rows = run_block(k, out, ctx, emit)
 716          rec['gate'] = g
 717          blog.write(json.dumps(rec) + '\n')
 718          blog.flush()
 719          print(f'[block {k}] {rec["label"]}' + (f': {"; ".join(rec["problems"])}' if rec['problems'] else '') +
 720                f'; {len(rows)} rows', flush=True)
 721          if rec.get('interrupted'):  # a signal: the record is kept, then the run stops
 722              raise SystemExit(f'interrupted in block {k}')
 723          if rec['label'] == 'INVALID':
 724              return f'block {k} INVALID: redo in a later session (--resume {out.name})'
 725      return 'all blocks done' if ctx['smoke'] or not next_blocks(read_jsonl(out / 'blocks.jsonl')) else 'blocks remain'

 170  def first_ids(post, text, ids, names, sep=''):
 171      """({name: first token id after text}, problems): text's ids must stay a prefix; first tokens distinct."""

 211  def score(post, it, v, order, prompt):
 212      """One row: probabilities over the offered names (raw and renormalised), top, margin, counters; V3 adds the
 213      reason and its step-1 counters."""
 214      names, (text, ids, nid) = offered(it, order), prompt
 215      t0, row = time.perf_counter(), {'variant': v, 'order': order, 'id': it['id'], 'offered': names}
 216      if v == 'V3':
 217          r1 = post('/completion', {'prompt': ids, **b0.DECODE, **REASON, 'cache_prompt': True})
 218          reason = r1['content']
 219          t2 = text + reason + ACTION
 220          ids = b0.tokenize(post, t2)
 221          nid, why = first_ids(post, t2, ids, names, ' ')
 222          if why:
 223              raise RuntimeError(f'scored position after the reason: {why}')
 224          tm1 = r1.get('timings', {})
 225          row.update(reason=reason, reason_tokens=r1.get('tokens_predicted'), reason_stop=r1.get('stop_type'),
 226                     reason_prompt_n=tm1.get('prompt_n'), reason_cache_n=tm1.get('cache_n'))
 227      body = {'prompt': ids, **b0.DECODE, **b0.PICK, 'cache_prompt': True}
 228      res = post('/completion', body)
 229      raw, norm = b0.probs(res, {n: [nid[n]] for n in names})
 230      if norm is None:
 231          raise RuntimeError(f'item {it["id"]}: an offered name is outside the top {b0.PICK["n_probs"]}: {raw}')
 232      top = max(norm, key=norm.get)  # ties: first offered
 233      second = sorted(norm.values())[-2] if len(norm) > 1 else 0.0
 234      tm = res.get('timings', {})

 236                 cache_n=tm.get('cache_n'), prompt_ms=tm.get('prompt_ms'), wall_s=round(time.perf_counter() - t0, 3),
 237                 probs_raw=raw, probs=norm, top=top, margin=norm[top] - second)

 109  def probs(res, ids_by_name):
 110      """Raw probabilities of the names' tokens at the first generated position (picker.Gemma.decide), None outside
 111      the top n_probs; renormalised over the five only when all are present."""
 112      by_id = {e['id']: (e['prob'] if 'prob' in e else math.exp(e['logprob']))

 116      norm = {n: v / mass for n, v in raw.items()} if None not in raw.values() and mass > 0 else None

 595  def pin_sampler():
 596      """Ask for the calling (sampler) thread on SAMPLER_CPUS at nice SAMPLER_NICE (both per thread on Linux; its su
 597      clients inherit them) and record what it actually got: a refused or ignored request is recorded, never assumed."""
 598      tid = threading.get_native_id()
 599      st = {'tid': tid, 'requested_cpus': list(SAMPLER_CPUS), 'requested_nice': SAMPLER_NICE}
 600      try:
 601          os.sched_setaffinity(0, SAMPLER_CPUS)
 602      except OSError as e:
 603          st['affinity_error'] = str(e)
 604      try:
 605          os.setpriority(os.PRIO_PROCESS, tid, SAMPLER_NICE)
 606      except OSError as e:
 607          st['nice_error'] = str(e)
 608      st.update(cpus=sorted(os.sched_getaffinity(0)), nice=os.getpriority(os.PRIO_PROCESS, tid))
 609      return st

 148      def check_affinity(self):
 149          """Re-pin and verify any of the 3 threads whose affinity drifted (Android cpuset changes can reset it).
 150          Never raises: the result says ok or not."""
 151          before = self.affinity()
 152          repin = [tid for tid, cpus in before.items() if isinstance(cpus, list) and set(cpus) != self.cpus]
 153          errors = {tid: cpus for tid, cpus in before.items() if isinstance(cpus, str)}
 154          for tid in repin:

 159          after = self.affinity() if repin else before
