  92  def name_ids(post, prompt, ids, names):
  93      """({name: ids after the prompt}, problems): each name must add exactly one token to the prompt's ids, the five
  94      distinct. A problem STOPs the run before any inference."""
  95      tails, why = {}, []
  96      for n in names:
  97          full = tokenize(post, prompt + n)
  98          tails[n] = full[len(ids):]
  99          if full[:len(ids)] != ids:
 100              why.append(f'{n!r}: prompt ids change when the name follows')
 101          elif len(tails[n]) != 1:
 102              why.append(f'{n!r}: {len(tails[n])} tokens after the model turn {tails[n]}, not one')
 103      firsts = [t[0] for t in tails.values() if t]
 104      if len(set(firsts)) != len(firsts):
 105          why.append(f'first tokens not distinct: {tails}')
 106      return tails, why

 109  def probs(res, ids_by_name):
 110      """Raw probabilities of the names' tokens at the first generated position (picker.Gemma.decide), None outside
 111      the top n_probs; renormalised over the five only when all are present."""
 112      by_id = {e['id']: (e['prob'] if 'prob' in e else math.exp(e['logprob']))
 113               for e in res['completion_probabilities'][0]['top_logprobs']}
 114      raw = {n: by_id.get(i[0]) for n, i in ids_by_name.items()}
 115      mass = sum(v for v in raw.values() if v is not None)
 116      norm = {n: v / mass for n, v in raw.items()} if None not in raw.values() and mass > 0 else None
 117      return raw, norm

 127  def run_sequence(post, s, pid, emit):
 128      """Checks, then R1..R8 and the FRESH repeats, in order; returns (rows, info). Stops at the first problem or error;
 129      an exception is recorded in info['exception'] and the rows collected so far are kept (the criteria then FAIL)."""
 130      rows, info = [], {'problems': []}
 131      try:
 132          sequence(post, s, pid, emit, rows, info)
 133      except Exception as e:
 134          info['exception'] = f'{type(e).__name__}: {e}'
 135          info['stopped'] = f'exception after {len(rows)} requests: {info["exception"]}'
 136      return rows, info

 139  def sequence(post, s, pid, emit, rows, info):
 140      msgs0 = messages(s, 'picker', 0)
 141      p0, ids0, bad = render(post, msgs0)
 142      info['name_ids'], why = name_ids(post, p0, ids0, s['names'])
 143      info['problems'] += why + ([bad] if bad else [])
 144      _, ids4, _ = render(post, messages(s, 'picker', 4))
 145      info['t4_tail_tokens_empty_reply'] = len(ids4)
 146      if not SIZE[0] <= len(ids4) <= SIZE[1]:
 147          info['problems'].append(f'T(4) + tail = {len(ids4)} tokens with an empty reply, not in {SIZE}')
 148      props = post('/props', None)
 149      info['n_ctx'] = props.get('default_generation_settings', {}).get('n_ctx')
 150      if info['n_ctx'] != CTX:
 151          info['problems'].append(f'server n_ctx {info["n_ctx"]!r} != {CTX}')
 152      info['parts'] = {'sys+catalogue': len(tokenize(post, '\n'.join([s['sys'], *s['catalogue']]), False)),
 153                       'mission+memory': len(tokenize(post, '\n'.join([s['mission'], *s['memory']]), False)),
 154                       'tail': len(tokenize(post, s['tail'], False)), 'speech': len(tokenize(post, s['speech'], False)),
 155                       'companion_sys': len(tokenize(post, s['companion_sys'], False))}
 156      # C2's reference: BOS + '<|turn>system\n' + T(1), i.e. the R4 prompt up to the end of E1
 157      info['t1_prefix_tokens'] = len(tokenize(post, '<|turn>system\n' + transcript(s, 1)))
 158      if info['problems']:
 159          info['stopped'] = 'STOP before any inference: ' + '; '.join(info['problems'])
 160          return
 161      slot, reply, sent = None, '', {}
 162      plan = [(r, kind, k, True) for r, kind, k in SEQUENCE] + [(r, None, None, False) for r in FRESH]
 163      for rid, kind, k, cache in plan:
 164          if cache:
 165              prompt, ids, bad = render(post, messages(s, kind, k, reply))
 166              if bad:
 167                  info['stopped'] = f'{rid}: {bad}'
 168                  break
 169              sent[rid] = (kind, k, ids)
 170          else:
 171              kind, k, ids = sent[rid]
 172          body = {'prompt': ids, **DECODE, **(PICK if kind == 'picker' else CHAT), 'cache_prompt': cache}
 173          row = {'id': rid if cache else rid + '_FRESH', 'kind': kind, 'k': k, 'cache_prompt': cache,
 174                 'prompt_tokens': len(ids), 'prompt_ids_sha256': sha(json.dumps(ids)),
 175                 'lcp_prev_slot': mem1.lcp(ids, slot) if slot is not None else None,
 176                 'pid': pid, 'pid_start': pid_start(pid)}
 177          if kind == 'picker' and not cache:
 178              row['probs_cached_ref'] = rid
 179          t0 = time.perf_counter()
 180          try:
 181              res = post('/completion', body)
 182          except Exception as e:  # fail closed: recorded, the sequence ends
 183              row.update(error=f'{type(e).__name__}: {e}', wall_s=round(time.perf_counter() - t0, 3))
 184              rows.append(row)
 185              emit(row)
 186              info['stopped'] = f'{row["id"]} failed: {e!r}'
 187              break
 188          tm, gen = res.get('timings', {}), res.get('tokens') or []
 189          row.update(wall_s=round(time.perf_counter() - t0, 3), prompt_n=tm.get('prompt_n'), cache_n=tm.get('cache_n'),
 190                     prompt_ms=tm.get('prompt_ms'), generated_ids=gen, content=res.get('content'),
 191                     tokens_cached=res.get('tokens_cached'), timings=tm)
 192          if kind == 'picker':
 193              row['probs_raw'], row['probs'] = probs(res, info['name_ids'])
 194              row['top'] = max(row['probs'], key=row['probs'].get) if row['probs'] else None
 195          if rid == 'R4':
 196              reply = res.get('content') or ''
 197              info['reply'] = reply
 198          slot = ids + gen
 199          rows.append(row)
 200          emit(row)

 203  def criteria(rows, info, exit_code, killed, exception=None):
 204      """[(name, PASS|FAIL|INFO, numbers)], from the recorded rows alone; any exception (run_sequence's or the
 205      runner's) fails C5, missing rows fail C1, C2 and C4."""
 206      r = {x['id']: x for x in rows}
 207      out = []
 208  
 209      def got(rid, key):
 210          return r.get(rid, {}).get(key)
 211      c1 = []
 212      for rid in ('R2', 'R3', 'R5', 'R8'):
 213          n, lcp, cn = got(rid, 'prompt_tokens'), got(rid, 'lcp_prev_slot'), got(rid, 'cache_n')
 214          need = min(lcp, n - 1) if None not in (n, lcp) else None
 215          c1.append((rid, cn, need, cn is not None and need is not None and cn >= need))
 216      out.append(('C1', 'PASS' if all(x[3] for x in c1) else 'FAIL',
 217                  '; '.join(f'{a}: cache_n {b} vs shared {c}' for a, b, c, _ in c1)))
 218      cn, need = got('R5', 'cache_n'), info.get('t1_prefix_tokens')
 219      out.append(('C2', 'PASS' if None not in (cn, need) and cn >= need else 'FAIL',
 220                  f'R5 cache_n {cn} vs T(1) without tail {need}'))
 221      out.append(('C3', 'INFO', f'R7 prompt_n {got("R7", "prompt_n")} (rebuild cost after the separate chat; '
 222                  f'cache_n {got("R7", "cache_n")}, prompt {got("R7", "prompt_tokens")})'))
 223      c4, dps = [], []
 224      for rid in FRESH:
 225          a, b = got(rid, 'probs'), got(rid + '_FRESH', 'probs')
 226          if a is None or b is None:
 227              c4.append((rid, None, None, False))
 228              continue
 229          dp = max(abs(a[n] - b[n]) for n in a)
 230          dps.append(dp)
 231          c4.append((rid, got(rid, 'top'), got(rid + '_FRESH', 'top'), got(rid, 'top') == got(rid + '_FRESH', 'top')))
 232      ok4 = all(x[3] for x in c4) and len(dps) == len(FRESH) and max(dps) <= MAX_DP
 233      out.append(('C4', 'PASS' if ok4 else 'FAIL', (f'max |dp| {max(dps):.6f}; ' if dps else 'no pairs; ') +
 234                  '; '.join(f'{a}: cached {b} / fresh {c}' for a, b, c, _ in c4)))
 235      pids = {(x['pid'], x['pid_start']) for x in rows}
 236      errors = sum('error' in x for x in rows)
 237      exc = exception or info.get('exception')
 238      ok5 = len(pids) == 1 and errors == 0 and exit_code == 0 and not killed and not exc and \
 239          len(rows) == len(SEQUENCE) + len(FRESH)
 240      out.append(('C5', 'PASS' if ok5 else 'FAIL', f'PIDs {sorted(pids)}; {len(rows)} requests, {errors} errors; '
 241                  f'exit code {exit_code}, SIGKILL {killed}; exception {exc}'))
 242      return out

 269                       f'  [{mass:.4f}]')
 270      L += ['', 'criteria:'] + [f'  {c}  {v:<4}  {t}' for c, v, t in crit]
 271      L.append(f'server exit code {run.get("exit_code")}; SIGKILL needed {run.get("killed")}')
 272      return '\n'.join(L) + '\n'

 275  def main():
 276      s_bytes = SET.read_bytes()
 277      s = json.loads(s_bytes)
 278      if bad := mem1.pinned_problems()[0]:
 279          raise SystemExit(f'refused: {bad}')
 280      if not repro.port_free():
 281          raise SystemExit('refused: port 8080 in use (never stops a server it did not start)')
 282      utc = time.strftime('%Y%m%dT%H%M%SZ', time.gmtime())
 283      out = OUT_ROOT / utc
 284      out.mkdir(parents=True)
 285      help_text = subprocess.run([mem1.SERVER, '--help'], capture_output=True, text=True, timeout=60).stdout
 286      run = {'utc': utc, 'set_sha256': sha(s_bytes), 'code_sha256': sha(Path(__file__).read_bytes()),
 287             'mem1_sha256': sha(Path(mem1.__file__).read_bytes()), 'pinned_sha256': mem1.PINNED,
 288             'git_head': subprocess.run(['git', '-C', str(HERE), 'rev-parse', 'HEAD'], capture_output=True,
 289                                        text=True).stdout.strip(),
 290             'command': mem1.command(CTX), 'decode': DECODE, 'picker': PICK, 'chat': CHAT,
 291             'slot_save_path_listed': '--slot-save-path' in help_text, 'timings': 'NOT MEASUREMENTS'}
 292      (out / 'run.json').write_text(json.dumps(run, indent=1) + '\n')  # set SHA-256 on disk before the server starts
 293  
 294      def post(path, body):
 295          if body is None:
 296              return mem1.props_json()
 297          return cr.post(path, body, timeout=REQUEST_TIMEOUT_S)
 298  
 299      with open(out / 'requests.jsonl', 'x') as rq, open(out / 'server.log', 'x') as log:
 300          def emit(row):
 301              rq.write(json.dumps(row) + '\n')
 302              rq.flush()
 303          killed, rows, info, proc, exc = False, [], {}, None, None
 304          try:
 305              proc = subprocess.Popen(mem1.command(CTX), stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT,
 306                                      start_new_session=True)
 307              t0 = time.monotonic()
 308              while not cr.healthy():
 309                  if proc.poll() is not None or time.monotonic() - t0 > LOAD_TIMEOUT_S:
 310                      raise RuntimeError(f'llama-server not healthy (exit {proc.poll()})')
 311                  time.sleep(0.1)
 312              run['server_pid'], run['server_comm'] = proc.pid, Path(f'/proc/{proc.pid}/comm').read_text().strip()
 313              rows, info = run_sequence(post, s, proc.pid, emit)
 314          except BaseException as e:  # recorded; the report and manifest are still written, then a stop re-raises
 315              exc = e
 316              run['exception'] = f'{type(e).__name__}: {e}'
 317          finally:
 318              if proc is not None and proc.poll() is None:
 319                  os.killpg(proc.pid, signal.SIGTERM)
 320                  try:
 321                      proc.wait(timeout=TERM_WAIT_S)
 322                  except subprocess.TimeoutExpired:
 323                      os.killpg(proc.pid, signal.SIGKILL)
 324                      proc.wait()
 325                      killed = True
 326              run.update(exit_code=proc.returncode if proc else None, killed=killed, info=info)
 327      print(finish(out, run, rows, info) + f'\n{out}')
 328      if exc is not None and not isinstance(exc, Exception):
 329          raise exc
 330  
 331  
 332  def finish(out, run, rows, info):
 333      """Criteria, run.json, report.txt and artifacts.sha256, always (also after an exception); returns the report."""
 334      run['criteria'] = criteria(rows, info, run.get('exit_code'), run.get('killed'), run.get('exception'))
 335      (out / 'run.json').write_text(json.dumps(run, indent=1) + '\n')
 336      text = report(run, rows, info, run['criteria'])
 337      (out / 'report.txt').write_text(text)
 338      (out / 'artifacts.sha256').write_text(''.join(f'{sha(p.read_bytes())}  {p.name}\n' for p in sorted(out.iterdir())
 339                                                    if p.name != 'artifacts.sha256'))
 340      return text

 166  def criteria(launch, rows, info, exit_code, killed, exception=None, reference=None):
 167      """[(name, PASS|FAIL, numbers)] for this launch, from the recorded rows alone."""
 168      r = {x['id']: x for x in rows}
 169      out = []
 170  
 171      def got(rid, key):
 172          return r.get(rid, {}).get(key)
 173  
 174      def reuse(rid):  # lcp_prev_slot is already against the right content (restored, or the request's own slot)
 175          n, lcp, cn = got(rid, 'prompt_tokens'), got(rid, 'lcp_prev_slot'), got(rid, 'cache_n')
 176          need = min(lcp, n - 1) if None not in (n, lcp) else None
 177          return cn, need, None not in (cn, need) and cn >= need
 178      if launch == 'L0':
 179          ref = {x['id']: x for x in reference or []}
 180          diff = [(i, f) for i in [x['id'] for x in reference or []] for f in K0_FIELDS
 181                  if r.get(i, {}).get(f) != ref[i].get(f)]
 182          ok = len(ref) == 13 and len(rows) == 13 and not diff
 183          out.append(('K0', 'PASS' if ok else 'FAIL', f'{len(rows)} rows vs {len(ref)} B0 rows; differing (id, field): '
 184                      f'{diff[:8] or "none"}'))
 185      if launch == 'L1':
 186          acts = info.get('slot_actions', [])
 187          ok_act = [a['action'] for a in acts] == ['save', 'restore'] and \
 188              acts[1]['response'].get('n_restored', 0) > 0
 189          cn, need, ok = reuse('R7')
 190          out.append(('K1', 'PASS' if ok and ok_act else 'FAIL', f'R7 cache_n {cn} vs min(LCP with restored, n-1) {need}; '
 191                      f'save/restore ok {ok_act}'))
 192      if launch == 'L2':
 193          cn, need, ok = reuse('R7')
 194          out.append(('K2', 'PASS' if ok and got('R7', 'slot') == 0 else 'FAIL',
 195                      f'R7 cache_n {cn} vs min(LCP with slot 0, n-1) {need}'))
 196          cn9, n6 = got('R9', 'cache_n'), got('R6', 'prompt_tokens')
 197          out.append(('K3', 'PASS' if None not in (cn9, n6) and cn9 >= n6 else 'FAIL',
 198                      f'R9 cache_n {cn9} vs R6 prompt tokens {n6}'))
 199      c4 = b0.criteria(rows, info, exit_code, killed)[3]
 200      out.append(('K4', c4[1], c4[2]))
 201      pids = {(x['pid'], x['pid_start']) for x in rows}
 202      errors = sum('error' in x for x in rows)
 203      exc = exception or info.get('exception')
 204      misrouted = [x['id'] for x in rows if 'error' not in x and launch == 'L2' and x.get('slot_reported') != x['slot']]
 205      ok5 = len(pids) == 1 and errors == 0 and not exc and not misrouted and len(rows) == n_rows(launch) and \
 206          exit_code == 0 and not killed
 207      out.append(('K5', 'PASS' if ok5 else 'FAIL', f'PIDs {sorted(pids)}; {len(rows)}/{n_rows(launch)} requests, '
 208                  f'{errors} errors; misrouted {misrouted}; exit code {exit_code}, SIGKILL {killed}; exception {exc}'))
 209      return out

 292             'b0_sha256': sha(Path(b0.__file__).read_bytes()), 'mem1_sha256': sha(Path(mem1.__file__).read_bytes()),
 293             'b0_record_sha256': sha(B0_RECORD.read_bytes()), 'pinned_sha256': mem1.PINNED,

 301      def post(path, body):
 302          if body is None:  # GET
 303              with urllib.request.urlopen(f'http://127.0.0.1:{cr.PORT}{path}', timeout=30) as r:
 304                  return json.loads(r.read())
 305          return cr.post(path, body, timeout=b0.REQUEST_TIMEOUT_S)

 310              rq.flush()
 311          killed, rows, info, proc, exc = False, [], {}, None, None

 314                                      start_new_session=True)

 118  def inside_repo(path):
 119      p = Path(path).resolve()  # symlinks resolved (B1 review P1)
 120      return any(q.exists() and os.path.samefile(q, REPO) for q in [p, *p.parents])  # bind mounts: same inode

 123  def load_set(path, want, custody=True):
 124      """(set, bytes SHA-256): refused when custody and the path is inside the repo (checked before reading) or the
 125      hash differs. 'visit' in labels, current activity and allowed lists reads 'approach'."""
 126      if custody and inside_repo(path):
 127          raise SystemExit(f'refused: {path} is inside the repository (test-set custody)')
 128      data = Path(path).read_bytes()
 129      if want is not None and sha(data) != want:
 130          raise SystemExit(f'refused: {path} SHA-256 {sha(data)[:12]} != pinned {want[:12]}')
 131      s = json.loads(data)
 132      ren = {'visit': 'approach'}
 133      for it in s['items']:
 134          it['current'], it['strict'] = ren.get(it['current'], it['current']), ren.get(it['strict'], it['strict'])
 135          it['allowed'], it['also_ok'] = [ren.get(n, n) for n in it['allowed']], [ren.get(n, n) for n in it['also_ok']]
 136      return s, sha(data)

 170  def first_ids(post, text, ids, names, sep=''):
 171      """({name: first token id after text}, problems): text's ids must stay a prefix; first tokens distinct."""
 172      out, why = {}, []
 173      for n in names:
 174          full = b0.tokenize(post, text + sep + n)
 175          if full[:len(ids)] != ids or len(full) == len(ids):
 176              why.append(f'{n!r}: prompt ids change or no token follows')
 177          else:
 178              out[n] = full[len(ids)]
 179      if len(set(out.values())) != len(out):
 180          why.append(f'first tokens not distinct: {out}')
 181      return out, why

 184  def preflight(post, s, blocks, info, tag):
 185      """prompts[(variant, order, id)] = (text, ids, {name: id}); every prompt rendered and tokenized, no /completion.
 186      Problems go to info['problems']."""
 187      prompts, sizes = {}, {}
 188      for v, order in blocks:
 189          for it in s['items']:
 190              names = offered(it, order)
 191              text, ids, bad = b0.render(post, messages(s, it, v, order))
 192              if v == 'V3':  # scored position checked with an empty reason; rechecked per item with the real one
 193                  t2 = text + ACTION
 194                  nid, why = first_ids(post, t2, b0.tokenize(post, t2), names, ' ')
 195              else:
 196                  nid, why = first_ids(post, text, ids, names)
 197              info['problems'] += [f'{tag} {v}-{order} item {it["id"]}: {w}' for w in why + ([bad] if bad else [])]
 198              prompts[v, order, it['id']] = (text, ids, nid)
 199              sizes.setdefault(v, []).append(len(ids))
 200              info.setdefault('name_ids', {}).setdefault('V3' if v == 'V3' else 'V1/V2', {}).update(nid)
 201              if tag == 'DEV' and it['id'] == SHOW_ITEM and order == 'FWD':
 202                  info[f'prompt_{v}_dev_item{SHOW_ITEM}'] = post('/detokenize', {'tokens': ids})['content']
 203      for v in sizes:
 204          info.setdefault('system_tokens', {})[v] = len(b0.tokenize(post, SYSTEM[v], special=False))
 205          info.setdefault('prompt_tokens', {})[f'{tag} {v}'] = (min(sizes[v]), max(sizes[v]))
 206          if max(sizes[v]) + CTX_MARGIN > CTX:
 207              info['problems'].append(f'{tag} {v}: longest prompt {max(sizes[v])} + {CTX_MARGIN} > {CTX}')
 208      return prompts

 211  def score(post, it, v, order, prompt):
 212      """One row: probabilities over the offered names (raw and renormalised), top, margin, counters; V3 adds the
 213      reason and its step-1 counters."""
 214      names, (text, ids, nid) = offered(it, order), prompt
 215      t0, row = time.perf_counter(), {'variant': v, 'order': order, 'id': it['id'], 'offered': names}
 216      if v == 'V3':
 217          r1 = post('/completion', {'prompt': ids, **b0.DECODE, **REASON, 'cache_prompt': True})
 218          reason = r1['content']
 219          t2 = text + reason + ACTION
 220          ids = b0.tokenize(post, t2)
 221          nid, why = first_ids(post, t2, ids, names, ' ')
 222          if why:
 223              raise RuntimeError(f'scored position after the reason: {why}')
 224          tm1 = r1.get('timings', {})
 225          row.update(reason=reason, reason_tokens=r1.get('tokens_predicted'), reason_stop=r1.get('stop_type'),
 226                     reason_prompt_n=tm1.get('prompt_n'), reason_cache_n=tm1.get('cache_n'))
 227      body = {'prompt': ids, **b0.DECODE, **b0.PICK, 'cache_prompt': True}
 228      res = post('/completion', body)
 229      raw, norm = b0.probs(res, {n: [nid[n]] for n in names})
 230      if norm is None:
 231          raise RuntimeError(f'item {it["id"]}: an offered name is outside the top {b0.PICK["n_probs"]}: {raw}')
 232      top = max(norm, key=norm.get)  # ties: first offered
 233      second = sorted(norm.values())[-2] if len(norm) > 1 else 0.0
 234      tm = res.get('timings', {})
 235      row.update(prompt_tokens=len(ids), prompt_ids_sha256=sha(json.dumps(ids)), prompt_n=tm.get('prompt_n'),
 236                 cache_n=tm.get('cache_n'), prompt_ms=tm.get('prompt_ms'), wall_s=round(time.perf_counter() - t0, 3),
 237                 probs_raw=raw, probs=norm, top=top, margin=norm[top] - second)
 238      return row

 241  def block_metrics(rows, s, v):
 242      """B1's metrics for one variant: (FWD block, REV block, {'flips', 'verdict'})."""
 243      # ponytail: reuses b1.metrics by relabelling the variant as B1's arm A; own metrics if B1's layout ever changes
 244      m = b1.metrics([{**r, 'arm': 'A'} for r in rows if r.get('variant') == v and r.get('set') == s['tag']], s)
 245      return m['blocks']['A-FWD'], m['blocks']['A-REV'], m['arms']['A']

 248  def select(dev):
 249      """Fixed rule. dev[v] = (FWD, REV, arm) from block_metrics. Among variants with flips <= MAX_FLIPS the highest
 250      mean acceptable over both orders; tie -> V1, then V2, then V3. None qualifies: highest mean anyway, flag FLIPS."""
 251      mean = {v: (dev[v][0]['acceptable'] + dev[v][1]['acceptable']) / 2 for v in VARIANTS}
 252      ok = [v for v in VARIANTS if len(dev[v][2]['flips']) <= MAX_FLIPS]
 253      pool = ok or VARIANTS
 254      best = max(pool, key=lambda v: (mean[v], -VARIANTS.index(v)))
 255      return {'variant': best, 'flag': None if ok else 'FLIPS', 'rule': select.__doc__.split('. ', 1)[1],
 256              'dev': {v: {'FWD': {k: dev[v][0][k] for k in ('strict', 'acceptable')},
 257                          'REV': {k: dev[v][1][k] for k in ('strict', 'acceptable')},
 258                          'mean_acceptable': mean[v], 'flips': dev[v][2]['flips']} for v in VARIANTS}}

 261  def run_blocks(post, s, blocks, prompts, pid, emit, rows, info, first):
 262      """Score the blocks in order; False at the first request error (recorded, the run stops)."""
 263      for k, (v, order) in enumerate(blocks):
 264          if k or not first:
 265              time.sleep(PAUSE_S)
 266          t0 = time.monotonic()
 267          for it in s['items']:
 268              try:
 269                  row = score(post, it, v, order, prompts[v, order, it['id']])
 270              except Exception as e:  # fail closed: recorded, the run ends
 271                  row = {'variant': v, 'order': order, 'id': it['id'], 'error': f'{type(e).__name__}: {e}'}
 272              row['set'], row['pid'], row['pid_start'] = s['tag'], pid, b0.pid_start(pid)
 273              rows.append(row)
 274              emit(row)
 275              if 'error' in row:
 276                  info['stopped'] = f'{s["tag"]} {v}-{order} item {it["id"]} failed: {row["error"]}'
 277                  return False
 278          dur = round(time.monotonic() - t0, 1)
 279          info.setdefault('blocks', []).append({'block': f'{s["tag"]} {v}-{order}', 'duration_s': dur,
 280                                                'over_warn': dur > BLOCK_WARN_S})
 281      return True

 284  def run_all(post, dev, test_path, out, pid, emit):
 285      """DEV preflight and blocks, selection.json, then TEST (only now opened) for the selected variant.
 286      Returns (rows, info); info['test'] = TEST set and hash when it was opened."""
 287      rows, info = [], {'problems': []}
 288      try:
 289          dev['tag'] = 'DEV'
 290          prompts = preflight(post, dev, DEV_BLOCKS, info, 'DEV')
 291          n_ctx = post('/props', None).get('default_generation_settings', {}).get('n_ctx')
 292          info['n_ctx'] = n_ctx
 293          if n_ctx != CTX:
 294              info['problems'].append(f'server n_ctx {n_ctx!r} != {CTX}')
 295          if info['problems']:
 296              info['stopped'] = 'STOP before any inference: ' + '; '.join(info['problems'])
 297              return rows, info
 298          if not run_blocks(post, dev, DEV_BLOCKS, prompts, pid, emit, rows, info, True):
 299              return rows, info
 300          sel = select({v: block_metrics(rows, dev, v) for v in VARIANTS})
 301          (out / 'selection.json').write_text(json.dumps(sel, indent=1) + '\n')
 302          info['selection'], info['selection_sha256'] = sel, sha((out / 'selection.json').read_bytes())
 303          emit({'event': 'selection', 'variant': sel['variant'], 'flag': sel['flag'],
 304                'selection_sha256': info['selection_sha256']})
 305          test, info['test_sha256'] = load_test(out, info['selection_sha256'], test_path)
 306          test['tag'], info['test'] = 'TEST', test
 307          blocks = [(sel['variant'], 'FWD'), (sel['variant'], 'REV')]
 308          prompts = preflight(post, test, blocks, info, 'TEST')
 309          if info['problems']:
 310              info['stopped'] = 'STOP before TEST inference: ' + '; '.join(info['problems'])
 311              return rows, info
 312          run_blocks(post, test, blocks, prompts, pid, emit, rows, info, False)
 313      except Exception as e:
 314          info['exception'] = f'{type(e).__name__}: {e}'
 315          info['stopped'] = f'exception after {len(rows)} requests: {info["exception"]}'
 316      return rows, info

 319  def validity(rows, info, run):
 320      why = []
 321      n = len(DEV_BLOCKS) * run['dev_rows'] + 2 * run.get('test_rows', 40)
 322      if len(rows) != n or any('error' in r for r in rows):
 323          why.append(f'{len(rows)}/{n} rows, {sum("error" in r for r in rows)} errors')
 324      if len({(r['pid'], r['pid_start']) for r in rows}) > 1:
 325          why.append('more than one server PID')
 326      if run.get('exit_code') != 0 or run.get('killed'):
 327          why.append(f'server exit {run.get("exit_code")}, SIGKILL {run.get("killed")}')
 328      for k in ('exception', 'stopped'):
 329          if info.get(k) or run.get(k):
 330              why.append(f'{k}: {info.get(k) or run.get(k)}')
 331      for k in ('dev', 'test'):
 332          if run.get(f'{k}_sha256_end') != run.get(f'{k}_sha256'):
 333              why.append(f'{k} SHA-256 at the end {run.get(f"{k}_sha256_end")} != start {run.get(f"{k}_sha256")}')
 334      return ('VALID' if not why else 'INVALID'), why

 337  def table(L, title, rows, s, variants):
 338      L += ['', title, 'block      strict        acceptable    copy rate      near-ties (margin < 0.05)']
 339      res = {}
 340      for v in variants:
 341          F, R, arm = res[v] = block_metrics(rows, s, v)
 342          for o, x in (('FWD', F), ('REV', R)):
 343              L.append(f'{v}-{o:<7} {x["strict"]:.3f} {round(x["strict"] * x["n"]):>2}/{x["n"]}  '
 344                       f'{x["acceptable"]:.3f} {round(x["acceptable"] * x["n"]):>2}/{x["n"]}  '
 345                       f'{(x["copy_rate"] or 0):.3f} {x["copy"][0]:>2}/{x["copy"][1]}  {x["near_ties"]}')
 346      L.append('recall per strict label (hits/n):')
 347      L += [f'  {v}-{o:<4} ' + '  '.join(f'{a} {h}/{n}' for a, (h, n) in x['recall'].items())
 348            for v in variants for o, x in zip(('FWD', 'REV'), res[v][:2])]
 349      L.append('confusion (rows strict, columns chosen):')
 350      L += [f'  {v}-{o:<4} ' + '; '.join(f'{a} -> ' + ', '.join(f'{c} {n}' for c, n in sorted(r.items()))
 351                                         for a, r in sorted(x['confusion'].items()))
 352            for v in variants for o, x in zip(('FWD', 'REV'), res[v][:2])]
 353      L.append('copied current activity (ids): ' + '; '.join(f'{v}-{o} {x["copied_ids"]}' for v in variants
 354                                                             for o, x in zip(('FWD', 'REV'), res[v][:2])))
 355      L += [f'flips {v}: {len(res[v][2]["flips"])} {res[v][2]["flips"]}' for v in variants]
 356      by = {(r['variant'], r['order'], r['id']): r for r in rows if r.get('set') == s['tag']}
 357      L.append('id  strict (also ok)       ' + ''.join(f'{v}-{o:<12}' for v in variants for o in ('FWD', 'REV')))
 358      for it in s['items']:
 359          cells = []
 360          for v in variants:
 361              for o in ('FWD', 'REV'):
 362                  r = by.get((v, o, it['id']), {})
 363                  cells.append(f'{r["top"]} {r["probs"][r["top"]]:.2f}' if 'top' in r else r.get('error', '-')[:14])
 364          lab = it['strict'] + (f' ({",".join(it["also_ok"])})' if it['also_ok'] else '')
 365          L.append(f'{it["id"]:<3} {lab:<22} ' + ''.join(f'{c:<15}' for c in cells))
 366      reasons = [r for r in rows if r.get('set') == s['tag'] and 'reason' in r]
 367      if reasons:
 368          L.append('V3 reasons (id order: top | stop | reason):')
 369          L += [f'  {r["order"]} {r["id"]:<3} {r.get("top", "-"):<8} {r.get("reason_stop")} | {r["reason"]!r}'
 370                for r in reasons]
 371      return res

 374  def report(run, rows, info):
 375      s_label, why = run['validity']
 376      L = [f'MIND-B2 {run["utc"]}  code {run["code_sha256"][:12]}  base {run["git_head"]}',
 377           f'DEV {run["dev_sha256"][:12]} ({run["dev_rows"]} rows, {run["dev_path"]}); TEST '
 378           f'{(run.get("test_sha256") or "not opened")[:12]} ({run["test_path"]})',
 379           f'server: {" ".join(run["command"])}',
 380           f'n_ctx {info.get("n_ctx")}; system tokens {info.get("system_tokens")}; prompt tokens (min, max) '
 381           f'{info.get("prompt_tokens")}',
 382           f'name first-token ids {info.get("name_ids")}',
 383           f'ACCURACY {s_label}' + (f': {"; ".join(why)}' if why else ' (temperature 0; 320/320 rows; one server PID; '
 384                                    'exit 0; set hashes unchanged)'),
 385           '* timings NOT MEASUREMENTS (agent resident allowed)']
 386      L += [f'block {x["block"]}: {x["duration_s"]} s*' + ('  OVER ~4 min (continued)' if x['over_warn'] else '')
 387            for x in info.get('blocks', [])]
 388      dev = {'tag': 'DEV', 'items': run['dev_items']}
 389      if any(r.get('set') == 'DEV' for r in rows):
 390          table(L, '=== DEV ===', rows, dev, VARIANTS)
 391      if sel := info.get('selection'):
 392          L += ['', f'SELECTION ({sel["rule"]}): {sel["variant"]}' + (f'  FLAG {sel["flag"]}' if sel['flag'] else '')
 393                + f'; selection.json SHA-256 {info["selection_sha256"]}']
 394          L += [f'  {v}: mean acceptable {d["mean_acceptable"]:.3f}, flips {len(d["flips"])}'
 395                for v, d in sel['dev'].items()]
 396      if info.get('test'):
 397          t = info['test']
 398          res = table(L, f'=== TEST (official; {sel["variant"]}) ===', rows, t, [sel['variant']])
 399          F, R, arm = res[sel['variant']]
 400          verdict = arm['verdict'] if s_label == 'VALID' else 'NO RESULT (run INVALID)'
 401          L += ['', f'PASS rule: acceptable >= {b1.PASS["acceptable"]} and strict >= {b1.PASS["strict"]} in both orders, '
 402                    f'flips <= {b1.PASS["max_flips"]}/40',
 403                f'  TEST {sel["variant"]}: strict {F["strict"]:.3f}/{R["strict"]:.3f}, acceptable '
 404                f'{F["acceptable"]:.3f}/{R["acceptable"]:.3f}, flips {len(arm["flips"])} {arm["flips"]} -> {verdict}']
 405      if f'prompt_V3_dev_item{SHOW_ITEM}' in info:
 406          L += ['', f'--- rendered prompt, V3 FWD, DEV item {SHOW_ITEM} (/detokenize) ---',
 407                info[f'prompt_V3_dev_item{SHOW_ITEM}']]
 408      if info.get('stopped'):
 409          L.append(f'STOPPED: {info["stopped"]}')
 410      if run.get('exception'):
 411          L.append(f'EXCEPTION (runner): {run["exception"]}')
 412      L.append(f'server exit code {run.get("exit_code")}; SIGKILL needed {run.get("killed")}')
 413      return '\n'.join(L) + '\n'

 416  def finish(out, run, rows, info):
 417      """Validity, run.json, report.txt and artifacts.sha256, always (also after an exception)."""
 418      run['validity'] = validity(rows, info, run)
 419      (out / 'run.json').write_text(json.dumps({**run, 'info': {k: v for k, v in info.items() if k != 'test'}},
 420                                               indent=1) + '\n')
 421      text = report(run, rows, info)
 422      (out / 'report.txt').write_text(text)
 423      (out / 'artifacts.sha256').write_text(''.join(f'{sha(p.read_bytes())}  {p.name}\n' for p in sorted(out.iterdir())
 424                                                    if p.name != 'artifacts.sha256'))
 425      return text

 428  def main():
 429      ap = argparse.ArgumentParser()
 430      ap.add_argument('--test', default=str(TEST))
 431      a = ap.parse_args()
 432      if inside_repo(a.test):  # custody: refused before anything runs; the file itself is opened only after selection
 433          raise SystemExit(f'refused: {a.test} is inside the repository (test-set custody)')
 434      dev, dev_sha = load_set(DEV, DEV_SHA256, custody=False)
 435      if bad := mem1.pinned_problems()[0]:
 436          raise SystemExit(f'refused: {bad}')
 437      if not b0.repro.port_free():
 438          raise SystemExit('refused: port 8080 in use (never stops a server it did not start)')
 439      utc = time.strftime('%Y%m%dT%H%M%SZ', time.gmtime())
 440      out = OUT_ROOT / utc
 441      out.mkdir(parents=True)

 113  def preflight(post, s, b):
 114      """(gemma, prompts, info): every prompt of every block rendered, tokenized and checked; no /completion.
 115      prompts[(arm, order, id)] = ids (A) or (ids, {name: [id]}) (B). info['problems'] non-empty = STOP."""
 116      info = {'problems': []}
 117      g = picker.Gemma(post)
 118      letters = picker.LETTERS[:len(b['names'])]
 119      g.letter_ids = [g.encode(L)[0] for L in letters]  # picker builds four (its MODES); five actions here
 120      info['letter_ids'] = dict(zip(letters, g.letter_ids))
 121      if any(len(g.encode(L)) != 1 for L in letters) or len(set(g.letter_ids)) != len(letters):
 122          info['problems'].append(f'letters not single distinct tokens: {info["letter_ids"]}')
 123      prompts, sizes = {}, {'A': [], 'B': []}
 124      for arm, order in BLOCKS:
 125          for it in s['items']:
 126              names = offered(it, order)
 127              if arm == 'A':
 128                  ids = g.prompt_ids(state(s, it), options(b, names))
 129                  prompts[arm, order, it['id']] = ids
 130              else:
 131                  text, ids, bad = b0.render(post, messages(s, b, it, order))
 132                  name_ids, why = b0.name_ids(post, text, ids, names)
 133                  info['problems'] += [f'B {order} item {it["id"]}: {w}' for w in why + ([bad] if bad else [])]
 134                  prompts[arm, order, it['id']] = (ids, name_ids)
 135                  info.setdefault('name_ids', {}).update(name_ids)
 136              sizes[arm].append(len(ids))
 137              if it['id'] == SHOW_ITEM and order == 'FWD':
 138                  info[f'prompt_{arm}_item{SHOW_ITEM}'] = post('/detokenize', {'tokens': ids})['content']
 139      info['prompt_tokens'] = {a: (min(v), max(v)) for a, v in sizes.items()}
 140      info['n_ctx'] = post('/props', None).get('default_generation_settings', {}).get('n_ctx')
 141      if info['n_ctx'] != CTX:
 142          info['problems'].append(f'server n_ctx {info["n_ctx"]!r} != {CTX}')
 143      if max(max(v) for v in sizes.values()) >= CTX:
 144          info['problems'].append(f'a prompt does not fit ctx {CTX}: {info["prompt_tokens"]}')
 145      return g, prompts, info

 148  def score(post, g, s, b, it, arm, order, prompt):
 149      """One row: probabilities over the offered names (raw and renormalised), top, margin, counters."""
 150      names = offered(it, order)
 151      t0 = time.perf_counter()
 152      if arm == 'A':
 153          dist = g.decide(state(s, it), options(b, names))  # picker's own request (cache_prompt false)
 154          body, res = post.last
 155          if body['prompt'] != prompt:
 156              raise RuntimeError('picker.Gemma sent ids differing from the preflight')
 157          raw, norm = b0.probs(res, {n: [g.letter_ids[j]] for j, n in enumerate(names)})
 158          if max(abs(norm[n] - dist[c]) for n, c in zip(names, options(b, names))) > 1e-9:
 159              raise RuntimeError('letter probabilities differ from picker.Gemma.decide')
 160      else:
 161          ids, name_ids = prompt
 162          body = {'prompt': ids, **b0.DECODE, **b0.PICK, 'cache_prompt': True}
 163          res = post('/completion', body)
 164          raw, norm = b0.probs(res, {n: name_ids[n] for n in names})
 165          if norm is None:
 166              raise RuntimeError(f'item {it["id"]}: an offered name is outside the top {b0.PICK["n_probs"]}: {raw}')
 167      top = max(norm, key=norm.get)  # ties: first offered
 168      second = sorted(norm.values())[-2] if len(norm) > 1 else 0.0
 169      tm = res.get('timings', {})
 170      return {'arm': arm, 'order': order, 'id': it['id'], 'offered': names, 'prompt_tokens': len(body['prompt']),
 171              'prompt_ids_sha256': sha(json.dumps(body['prompt'])), 'cache_prompt': body['cache_prompt'],
 172              'prompt_n': tm.get('prompt_n'), 'cache_n': tm.get('cache_n'), 'prompt_ms': tm.get('prompt_ms'),
 173              'wall_s': round(time.perf_counter() - t0, 3), 'probs_raw': raw, 'probs': norm, 'top': top,
 174              'margin': norm[top] - second}

 177  def run_blocks(post, s, b, pid, emit):
 178      """Preflight, then the four blocks; returns (rows, info). Stops at the first problem or request error; an
 179      exception is recorded in info['exception'] and the rows collected so far are kept."""
 180      rows, info = [], {}
 181      try:
 182          g, prompts, info = preflight(post, s, b)
 183          if info['problems']:
 184              info['stopped'] = 'STOP before any inference: ' + '; '.join(info['problems'])
 185              return rows, info
 186          info['blocks'] = []
 187          for k, (arm, order) in enumerate(BLOCKS):
 188              if k:
 189                  time.sleep(PAUSE_S)
 190              t0 = time.monotonic()
 191              for it in s['items']:
 192                  try:
 193                      row = score(post, g, s, b, it, arm, order, prompts[arm, order, it['id']])
 194                  except Exception as e:  # fail closed: recorded, the run ends
 195                      row = {'arm': arm, 'order': order, 'id': it['id'], 'error': f'{type(e).__name__}: {e}'}
 196                  row['pid'], row['pid_start'] = pid, b0.pid_start(pid)
 197                  rows.append(row)
 198                  emit(row)
 199                  if 'error' in row:
 200                      info['stopped'] = f'{arm}-{order} item {it["id"]} failed: {row["error"]}'
 201                      return rows, info
 202              dur = round(time.monotonic() - t0, 1)
 203              info['blocks'].append({'block': f'{arm}-{order}', 'duration_s': dur, 'over_warn': dur > BLOCK_WARN_S})
 204      except Exception as e:
 205          info['exception'] = f'{type(e).__name__}: {e}'
 206          info['stopped'] = f'exception after {len(rows)} requests: {info["exception"]}'
 207      return rows, info

 210  def metrics(rows, s):
 211      """{'blocks': {'A-FWD': {...}}, 'arms': {'A': {...}}} from the recorded rows alone; denominators are the set's
 212      items (a missing row counts as wrong)."""
 213      items, n = {it['id']: it for it in s['items']}, len(s['items'])
 214      top = {(r['arm'], r['order'], r['id']): r['top'] for r in rows if 'top' in r}
 215      margin = {(r['arm'], r['order'], r['id']): r['margin'] for r in rows if 'top' in r}
 216      blocks, arms = {}, {}
 217      for arm, order in BLOCKS:
 218          got = {i: top.get((arm, order, i)) for i in items}
 219          strict = sum(got[i] == it['strict'] for i, it in items.items())
 220          ok = sum(got[i] in [it['strict'], *it['also_ok']] for i, it in items.items())
 221          recall, confusion = {}, {}
 222          for i, it in items.items():
 223              h, m = recall.get(it['strict'], (0, 0))
 224              recall[it['strict']] = (h + (got[i] == it['strict']), m + 1)
 225              row = confusion.setdefault(it['strict'], {})
 226              row[str(got[i])] = row.get(str(got[i]), 0) + 1
 227          copy_ids = [i for i, it in items.items() if it['strict'] != it['current']]
 228          copied = [i for i in copy_ids if got[i] == items[i]['current']]
 229          blocks[f'{arm}-{order}'] = {
 230              'n': n, 'rows': sum(v is not None for v in got.values()), 'strict': strict / n, 'acceptable': ok / n,
 231              'recall': {k: list(v) for k, v in sorted(recall.items())}, 'confusion': confusion,
 232              'copy_rate': len(copied) / len(copy_ids) if copy_ids else None, 'copy': [len(copied), len(copy_ids)],
 233              'copied_ids': copied,
 234              'near_ties': [i for i in items if margin.get((arm, order, i), 1.0) < NEAR_TIE]}
 235      for arm in ('A', 'B'):
 236          flips = [i for i in items if top.get((arm, 'FWD', i)) != top.get((arm, 'REV', i))]
 237          F, R = blocks[f'{arm}-FWD'], blocks[f'{arm}-REV']
 238          ok = min(F['acceptable'], R['acceptable']) >= PASS['acceptable'] and \
 239              min(F['strict'], R['strict']) >= PASS['strict'] and len(flips) <= PASS['max_flips']
 240          arms[arm] = {'flips': flips, 'verdict': 'PASS' if ok else 'FAIL'}
 241      return {'blocks': blocks, 'arms': arms}

 244  def validity(rows, info, run):
 245      why = []
 246      if len(rows) != len(BLOCKS) * run['set_rows'] or any('error' in r for r in rows):
 247          why.append(f'{len(rows)} rows, {sum("error" in r for r in rows)} errors')
 248      if len({(r['pid'], r['pid_start']) for r in rows}) > 1:
 249          why.append('more than one server PID')
 250      if run.get('exit_code') != 0 or run.get('killed'):
 251          why.append(f'server exit {run.get("exit_code")}, SIGKILL {run.get("killed")}')
 252      for k in ('exception', 'stopped'):
 253          if info.get(k) or run.get(k):
 254              why.append(f'{k}: {info.get(k) or run.get(k)}')
 255      if run.get('set_sha256_end') != run['set_sha256']:
 256          why.append(f'set SHA-256 at the end {run.get("set_sha256_end")} != start')
 257      return ('VALID' if not why else 'INVALID'), why
