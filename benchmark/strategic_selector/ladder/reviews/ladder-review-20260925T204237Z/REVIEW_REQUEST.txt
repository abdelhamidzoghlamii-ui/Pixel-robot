    97	def make_cold(files):
    98	    """Drop the page cache via root; if that is not possible, evict this model's weight files."""
    99	    r = subprocess.run(["su", "-c", "sync; echo 3 > /proc/sys/vm/drop_caches"], capture_output=True, text=True)
   100	    if r.returncode == 0:
   101	        return {"mode": "full-cold (drop_caches)"}
   102	    for f in files:
   103	        fd = os.open(f, os.O_RDONLY)
   104	        try:
   105	            os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)  # read-only files: every page is clean
   106	        finally:
   107	            os.close(fd)
   108	    reason = (r.stderr or r.stdout).strip().splitlines()[-1:] or [f"exit {r.returncode}"]
   109	    return {"mode": "weights-cold (posix_fadvise DONTNEED on the weight files; libraries stay cached)",
   110	            "drop_caches_error": reason[0]}

   146	        """EOF -> adapter.close(); returns ru_maxrss (MiB) of the largest process in the worker's tree."""
   147	        maxrss = None
   148	        try:
   149	            self.proc.stdin.close()

   156	            except ProcessLookupError:

   201	def fit(xs, ys):
   202	    """Least squares ms = fixed + per_token * tokens. None when the tokens have no spread."""
   203	    if len(set(xs)) < 2:
   204	        return None
   205	    slope, intercept = statistics.linear_regression(xs, ys)
   206	    r2 = statistics.correlation(xs, ys) ** 2 if len(set(ys)) > 1 else 0.0
   207	    return {"fixed_ms": intercept, "per_token_ms": slope, "r2": r2, "n": len(xs), "tokens_min": min(xs),
   208	            "tokens_max": max(xs)}

   215	def analyse(name, blocks, cases):
   216	    main = next(b for b in blocks if b["block"] == "main")
   217	    rows = main["rows"]
   218	    by = lambda field: {v: {"correct": rate([r for r in rows if r[field] == v], "correct"),
   219	                            "acceptable": rate([r for r in rows if r[field] == v], "acceptable")}
   220	                        for v in sorted({r[field] for r in rows})}
   221	    written = {r["case_id"]: r["choice"] for r in rows if r["order"] == "written"}
   222	    reverse = {r["case_id"]: r["choice"] for r in rows if r["order"] == "reversed"}
   223	    ms = [r["total_ms"] for r in rows]
   224	    levels = sorted({r["level"] for r in rows})
   225	    scaling = {b["threads"]: {"median_ms": median([r["total_ms"] for r in b["rows"]]), "n": len(b["rows"])}
   226	               for b in blocks if b["block"] != "main"}
   227	    main_written = {r["case_id"]: r for r in rows if r["order"] == "written"}
   228	    thread_choice_diffs = sum(r["choice"] != main_written[r["case_id"]]["choice"]
   229	                              for b in blocks if b["block"] != "main" for r in b["rows"])
   230	    thread_prob_diff = max([abs(r["distribution"][k] - main_written[r["case_id"]]["distribution"][k])
   231	                            for b in blocks if b["block"] != "main" for r in b["rows"] for k in r["distribution"]] or [0])
   232	    cached = next((b for b in blocks if b["block"] == f"t{main['threads']}"), None)
   233	    # Same decision, same thread count, two blocks: their ratio shows how disturbed the timings were.
   234	    repeat = [(r["case_id"], main_written[r["case_id"]]["total_ms"], r["total_ms"]) for r in (cached or {"rows": []})["rows"]]
   235	    worst = max(repeat, key=lambda x: max(x[1], x[2]) / min(x[1], x[2]), default=None)
   236	    load = lambda b: None if b is None else {**b["phases"], "total": sum(b["phases"].values()),
   237	                                              "spawn_to_ready": b["spawn_to_ready_ms"], "file_bytes": b["file_bytes"],
   238	                                              "state": b["load_state"]}
   239	    return {
   240	        "model": name, "threads_main": main["threads"], "cases": len(written),
   241	        "correct": rate(rows, "correct"), "acceptable": rate(rows, "acceptable"),
   242	        "by_level": by("level"), "by_phrasing": by("phrasing"),
   243	        "flips": sum(written[c] != reverse[c] for c in written), "flip_cases": [c for c in written if written[c] != reverse[c]],
   244	        "median_ms": median(ms), "p95_ms": p95(ms),
   245	        "tok_per_s": median([r["tokens"] / (r["total_ms"] / 1000) for r in rows]),
   246	        "split_ms": {k: median([r[k] for r in rows]) for k in ("tokenise_ms", "forward_ms", "post_ms")},
   247	        "split_share": {k: sum(r[k] for r in rows) / sum(ms) for k in ("tokenise_ms", "forward_ms", "post_ms")},
   248	        "fit": fit([r["tokens"] for r in rows], ms),
   249	        "tokens_by_level": {L: median([r["tokens"] for r in rows if r["level"] == L]) for L in levels},
   250	        "truncated": sum(r["longest"] >= r["limit"] for r in rows), "limit": rows[0]["limit"],
   251	        "scaling": scaling, "repeat_worst": worst and {"case_id": worst[0], "main_ms": worst[1], "repeat_ms": worst[2],
   252	                                                        "ratio": max(worst[1], worst[2]) / min(worst[1], worst[2])},
   253	        "thread_choice_diffs": thread_choice_diffs, "thread_max_prob_diff": thread_prob_diff,
   254	        "cold": load(main), "cached": load(cached),
   255	        "peak_mib": max(b["peak_mib"] for b in blocks), "wait4_maxrss_mib": max(b["wait4_maxrss_mib"] or 0 for b in blocks),

   262	def hints(m, mem_total_mib):
   263	    out = []
   264	    pct = lambda x: f"{x * 100:.0f}%"
   265	    f = m["fit"]
   266	    if f is None:
   267	        out.append(f"no latency-vs-length hint: all decisions had the same token count "
   268	                   f"({sorted(set(m['tokens_by_level'].values()))}), nothing to fit")
   269	    elif f["r2"] < 0.5:
   270	        out.append(f"latency is poorly explained by input length (R2 {f['r2']:.2f}, n {f['n']}, tokens "
   271	                   f"{f['tokens_min']}-{f['tokens_max']}) -> shorter input is not a reliable speed-up")
   272	    elif f["per_token_ms"] <= 0:
   273	        out.append(f"no measurable per-token cost (slope {f['per_token_ms']:.3f} ms/token, R2 {f['r2']:.2f}) "
   274	                   "-> shorter input will not help")
   275	    else:
   276	        top = max(m["tokens_by_level"])
   277	        t = m["tokens_by_level"][top]
   278	        share = f["per_token_ms"] * t / (f["fixed_ms"] + f["per_token_ms"] * t)
   279	        ev = (f"fit {f['fixed_ms']:.0f} ms + {f['per_token_ms']:.2f} ms/token x {t:.0f} tokens at level {top}, "
   280	              f"R2 {f['r2']:.2f}")
   281	        if share >= 0.5:
   282	            out.append(f"per-token cost is {pct(share)} of predicted latency at level {top} ({ev}) -> shorter input will help")
   283	        else:
   284	            out.append(f"fixed cost is {pct(1 - share)} of predicted latency at level {top} ({ev}) -> shortening input "
   285	                       "saves little; fewer model calls will help more")
   286	    s, ms = m["split_share"], m["split_ms"]
   287	    split_ev = (f"median tokenise {ms['tokenise_ms']:.1f} / forward {ms['forward_ms']:.1f} / post {ms['post_ms']:.1f} ms; "
   288	                f"shares {pct(s['tokenise_ms'])}/{pct(s['forward_ms'])}/{pct(s['post_ms'])}")
   289	    if s["tokenise_ms"] >= 0.2:
   290	        out.append(f"tokenisation is {pct(s['tokenise_ms'])} of decision time ({split_ev}) -> pre-tokenise constant "
   291	                   "parts or use a faster tokenizer")
   292	    if s["post_ms"] >= 0.2:
   293	        out.append(f"post-processing is {pct(s['post_ms'])} of decision time ({split_ev}) -> the glue code around the "
   294	                   "model is worth optimising")
   295	    if s["forward_ms"] >= 0.8:
   296	        out.append(f"model forward is {pct(s['forward_ms'])} of decision time ({split_ev}) -> only fewer tokens or a "
   297	                   "faster model will help; glue code is not the bottleneck")
   298	    sc = {t: v["median_ms"] for t, v in m["scaling"].items() if v["median_ms"] is not None}
   299	    rw = m["repeat_worst"]

    37	class Clock:
    38	    def __init__(self):
    39	        self.tok = self.fwd = 0.0
    40	        self._fwd_start = None
    41	
    42	    def timed(self, fn, field):
    43	        def run(*a, **k):
    44	            began = time.perf_counter()
    45	            try:
    46	                return fn(*a, **k)
    47	            finally:
    48	                setattr(self, field, getattr(self, field) + time.perf_counter() - began)
    49	        return run

    55	        def post(*_):
